874 call drv_input_data_get( indata, pint_fldn, 'ilev',pverp,recno, pint_ptrs )
875
876 do c=begchunk,endchunk
877 ncol = phys_state(c)%ncol
878 if (all(pmid_ptrs(c)%array(:ncol,:) > 0._r8 )) then
879 do i=1,ncol
880 lnpmid_ptrs(c)%array(i,:) = log(pmid_ptrs(c)%array(i,:))
881 lnpint_ptrs(c)%array(i,:) = log(pint_ptrs(c)%array(i,:))
882 enddo
883 endif
884
885 ! adjust temperatue input above tropopause for Fixed Dynamical Heating ....
886 if (do_fdh) then
887 pbuf => pbuf_get_chunk(pbuf2d, c)
888
889 call pbuf_get_field(pbuf, tropp_idx, tropp)
890 call pbuf_get_field(pbuf, tcorr_idx, tcorr)
891 call pbuf_get_field(pbuf, qrsin_idx, qrsin)
892 call pbuf_get_field(pbuf, qrlin_idx, qrlin)
893
894 call tropopause_find(phys_state(c), troplev, tropP=tropp(:), primary=TROP_ALG_HYBSTOB, backup=TROP_ALG_CLIMATE)
895
896 qrsin(:,:) = qrs_ptrs(c)%array(:,:)
897 qrlin(:,:) = qrl_ptrs(c)%array(:,:)
898
899 do k = 1, pver
900 do i = 1, ncol
901 if ( phys_state(c)%pmid(i,k) < tropp(i) ) then
902 temp_ptrs(c)%array(i,k) = temp_ptrs(c)%array(i,k) + tcorr(i,k)
903 endif
904 enddo
905 enddo
906 endif
907
908 enddo
909
910 call get_rad_cnst_data(indata, phys_state, pbuf2d, recno)
911
912 end subroutine rad_data_read
