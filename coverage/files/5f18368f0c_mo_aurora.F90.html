3!-----------------------------------------------------------------------
4!
5! Auroral oval parameterization. See reference:
6! R.G. Roble, E.C. Ridley
7! An auroral model for the NCAR thermospheric general circulation model (TGCM)
8! Annales Geophysicae,5A, (6), 369-382, 1987.
9!
10! The aurora oval is a circle in auroral circle coordinates. Auroral circle
11! coordinates are offset from magnetic coordinates by offa degrees (radians)
12! towards 0 MLT and by dskofa degrees (radians) towards dusk (18 MLT).
13! The aurora assumes a Maxwellian in energy, so that the characteristic
14! energy is half of the mean energy (or mean energy = 2*alfa, where alfa
15! is the characteristic energy). The Maxwellian is approximated in the
16! aion and bion subroutines.
17! The aurora oval is assumed to be a Gaussian in auroral latitude, with
18! peak values on the day (=1) and night (=2) sides that change from one to
19! the other using cosines of the auroral longitude coordinate.
20! There is provision for a low energy (~75 eV) aurora at the location of the
21! regular (~1-6 keV) aurora in order to simulate the energy flux found
22! at higher altitudes that is non-Maxwellian, but the flux is usually
23! set to zero (1.e-80).
24! There is provision for a proton (MeV) aurora, but the flux is usually
25! set to zero (1.e-20).
26! The drizzle is a constant low energy electron flux over the polar cap,
27! which goes to 1/e over twice the half-width of the aurora at the
28! radius of the aurora.
29! The cusp is a low energy electron flux centered over the dayside convection
30! entrance at phid at the convection reversal boundary theta0. The cusp
31! falls off over 5 degrees in latitude and over 20 degrees in longitude
32! to 1/e values of the peak at the center.
33! 1.e-20 and 1.e-80 are used to give a near zero answer.
34!
35! The polar drizzle and cusp electron energies are low, and soft particles
36! have great influence on the over-all thermospheric and ionospheric
37! structure, especially on the electron density profiles at mid-latitudes
38! and in winter since low energy electrons produce ionization at high
39! altitudes where loss rates are very low. (Comment by Wenbin Wang.)
40! The original energies for drizzle and cusp were alfad=0.75, alfac=0.5 keV.
41! The original guess at energy fluxes were: ed=0.1+2.0*power/100.,ec=0.1+0.9*power/100.
42! The next guess at energy fluxes were: ed=0.01+0.2*power/100., ec=0.01+0.09*power/100.
43! The values below reflect higher estimates for the electron energy (lower alt)
44!
45! Calling sequence (all subs in mo_aurora, mo_aurora.F):
46! 1) sub aurora_cons called once per time step from advance.
47! 2) sub aurora called from dynamics, inside parallel latitude scan.
48! 3) subs aurora_cusp and aurora_heat called from sub aurora.
49! 4) sub aurora_ions called from sub aurora.
50!

593 use ppgrid, only : pcols, pver
594

598! ... dummy arguments
599!-----------------------------------------------------------------------
600 integer, intent(in) :: &
601 ncol, & ! column count
602 lchnk ! chunk index
603 real(r8), intent(in) :: &
604 calday ! calendar day of year
605 real(r8), intent(in) :: &
606 tn(pcols,pver), & ! neutral gas temperature (K)
607 o2(ncol,pver), & ! O2 concentration (kg/kg)
608 o1(ncol,pver), & ! O concentration (kg/kg)
609 mbar(ncol,pver) ! mean molecular weight (g/mole)

616 rlons(ncol)
617 real(r8), intent(out) :: &
618 aur_hrate(ncol,pver) ! auroral heating rate
619
620!-----------------------------------------------------------------------
621! ... local variables
622!-----------------------------------------------------------------------
623 real(r8), parameter :: aur_therm = 807._r8
624 real(r8), parameter :: jkcal = 4184._r8
625 real(r8), parameter :: aur_heat_eff = .05_r8
626 real(r8), parameter :: aur_hconst = 1.e3_r8*jkcal*aur_therm*aur_heat_eff
627
628 integer :: i, k
629 integer :: hemis(ncol)
630 real(r8) :: r2d
631 real(r8) :: ofda, cosofa, sinofa, aslona
632 real(r8) :: sunlons(ncol) ! sun's mag longitudes
633 real(r8) :: dlat_aur(ncol)
634 real(r8) :: dlon_aur(ncol)

638 real(r8) :: coslon(ncol)

723 write(iulog,*) '-----------------------------------------------------'

728!-----------------------------------------------------------------------
729 call aurora_cusp( cusp, do_aurora, hemis, colat, alon, ncol )

733!-----------------------------------------------------------------------

744 ncol, lchnk )

750 end do

787 subroutine aurora_heat( flux, flux2, flux3, alfa, alfa2, &
788 alfa3, qteaur, drizl, do_aurora, hemis, &
789 alon, colat, ncol )
790!-----------------------------------------------------------------------
791! ... calculate alfa, flux, and drizzle
792!-----------------------------------------------------------------------
793
794 implicit none
795
796!-----------------------------------------------------------------------
797! ... dummy arguments
798!-----------------------------------------------------------------------
799 integer, intent(in) :: ncol
800 integer, intent(in) :: hemis(ncol)
801 real(r8), intent(in) :: colat(ncol)
802 real(r8), intent(in) :: alon(ncol)
803 real(r8), intent(inout) :: flux(ncol)
804 real(r8), intent(inout) :: flux2(ncol)
805 real(r8), intent(inout) :: flux3(ncol)
806 real(r8), intent(inout) :: drizl(ncol)
807 real(r8), intent(inout) :: qteaur(ncol)
808 real(r8), intent(inout) :: alfa(ncol)
809 real(r8), intent(inout) :: alfa2(ncol)
810 real(r8), intent(inout) :: alfa3(ncol)
811 logical, intent(in) :: do_aurora(ncol)
812
813!-----------------------------------------------------------------------
814! ... local variables
815!-----------------------------------------------------------------------
816 real(r8), dimension(ncol) :: &
817 coslamda, & ! cos(angle from throat)
818 halfwidth, & ! oval half-width
819 wrk, & ! temp wrk array
820 dtheta ! latitudinal variation (Gaussian)
821
822!-----------------------------------------------------------------------
823! Low-energy protons:
824!
825! alfap0 = 0.5*(alfap1+alfap2)
826! e0p = 0.5*(pe1+pe2)
827!
828! coslamda = cos(lamda)
829! halfwidth = auroral half width
830! dtheta = colat-theta0(ihem)
831! alfa = electron energy
832!-----------------------------------------------------------------------
833 where( do_aurora(:) )
834 coslamda(:) = cos( atan2( sin( alon(:) - rrote ),cos( alon(:) - rrote ) ) )
835!-----------------------------------------------------------------------
836! ... auroral oval half-width (equation (1) in Roble,1987):
837!-----------------------------------------------------------------------
838 halfwidth(:) = h0*(1._r8 - rh*cos( atan2( sin(alon(:) - rroth),cos( alon(:) - rroth ) ) ) )
839 dtheta(:) = colat(:) - rrad(hemis(:))
840 endwhere
841!-----------------------------------------------------------------------
842! ... characteristic energy (equation (2) in Roble,1987):
843!-----------------------------------------------------------------------
844 if( alfa0 > .01_r8 ) then
845 where( do_aurora(:) )
846 alfa(:) = alfa0*(1._r8 - ralfa*coslamda(:))
847 endwhere
848 else
849 alfa(:) = 0._r8
850 end if
851
852 where( do_aurora(:) )
853!-----------------------------------------------------------------------
854! ... flux, drizzle, alfa2, flux2
855!-----------------------------------------------------------------------
856!-----------------------------------------------------------------------
857! ... energy flux (equation (3) in Roble,1987):
858!-----------------------------------------------------------------------
859 wrk(:) = exp( -(dtheta(:)/halfwidth(:))**2 )
860 flux(:) = e0*(1._r8 - ree*coslamda(:))*wrk(:) / (2._r8*alfa(:)*1.602e-9_r8)
861 drizl(:) = exp( -((dtheta(:) + abs(dtheta(:)))/(2._r8*h0))**2 )
862 alfa2(:) = alfa20*(1._r8 - ralfa2*coslamda(:))
863 flux2(:) = e20*(1._r8 - re2*coslamda(:))*wrk(:) / (2._r8*alfa2(:)*1.602e-9_r8)
864!-----------------------------------------------------------------------
865! ... alfa3, flux3 for high energy electrons:
866!-----------------------------------------------------------------------
867 alfa3(:) = alfa30
868 flux3(:) = e30*wrk(:) / 1.602e-6_r8
869!-----------------------------------------------------------------------
870! ... for electron temperature (used in settei):
871!-----------------------------------------------------------------------
872 qteaur(:) = -7.e8_r8*wrk(:)
873 endwhere
874
875 end subroutine aurora_heat

877 subroutine aurora_ions( drizl, cusp, alfa1, alfa2, alfa3, &
878 flux1, flux2, flux3, tn, o2, &
879 o1, mbar, qo2p, qop, qn2p, &
880 qnp, pmid, do_aurora, ncol, lchnk, pbuf )
881!-----------------------------------------------------------------------
882! ... calculate auroral additions to ionization rates
883!-----------------------------------------------------------------------
884
885 use ppgrid, only : pcols, pver
886 use cam_history, only : outfld
887
888 use physics_buffer,only: physics_buffer_desc, pbuf_set_field, pbuf_get_field
889
890 implicit none
891
892!-----------------------------------------------------------------------
893! ... dummy arguments
894!-----------------------------------------------------------------------
895 integer, intent(in) :: ncol
896 integer, intent(in) :: lchnk
897 real(r8), intent(in), dimension(ncol) :: &
898 drizl, &
899 cusp, &
900 alfa1, &
901 alfa2, &
902 alfa3, &
903 flux1, &
904 flux2, &
905 flux3
906 real(r8), dimension(pcols,pver), intent(in) :: &
907 tn, & ! midpoint neutral temperature (K)
908 pmid ! midpoint pressure (Pa)
909 real(r8), dimension(ncol,pver), intent(in) :: &
910 o2, & ! midpoint o2 concentration (kg/kg)
911 o1, & ! midpoint o concentration (kg/kg)
912 mbar ! mean molecular mass (g/mole)
913 real(r8), dimension(ncol,pver), intent(inout) :: &
914 qo2p, & ! o2p prod from aurora (molecules/cm^3/s)
915 qop, & ! op prod from aurora (molecules/cm^3/s)
916 qn2p, & ! n2p prod from aurora (molecules/cm^3/s)
917 qnp ! np prod from aurora (molecules/cm^3/s)
918 logical, intent(in) :: do_aurora(ncol)
919
920 type(physics_buffer_desc),pointer :: pbuf(:)
921
922!-----------------------------------------------------------------------
923! ... local variables
924!-----------------------------------------------------------------------
925 real(r8), parameter :: const0 = 1.e-20_r8
926
927 integer :: i, k
928 real(r8), dimension(ncol) :: &
929 p0ez, &
930 press, & ! pressure at interface levels (dyne/cm^2)
931 tempi, & ! temperature at interface levels (K)
932 xalfa1, &
933 xalfa2, &
934 xcusp, &
935 xdrizl, & ! input to sub aion
936 xalfa_sp, &
937 xalfa3, &
938 flux1_ion, &
939 flux2_ion, &
940 cusp_ion, &
941 drizl_ion, & ! output from sub aion
942 alfa1_ion, &
943 alfa2_ion, &
944 alfa3_ion, & ! output from sub aion
945 alfasp_bion, & ! output from sub bion
946 barm_t, &
947 qsum, &
948 denom, &
949 p0ez_mbar, &
950 tk_mbar, &
951 barm, &
952 falfa1, &
953 falfa2, &
954 fcusp, &
955 fdrizl, &
956 falfa_sp, &
957 xn2, &
958 falfa3
959 real(r8), dimension(ncol) :: &
960 qo2p_aur, &
961 qop_aur, &
962 qn2p_aur ! auroral ionization for O2+, O+, N2+
963 real(r8) :: qia(5) ! low energy proton source (not in use, 1/02)
964 real(r8) :: wrk(ncol,pver)
965
966 real(r8), pointer :: aurIPRateSum(:,:) ! Pointer to pbuf auroral ion production sum for O2+,O+,N2+ (s-1 cm-3)
967
968 qia(:) = 0._r8
969 wrk(:,:) = 0._r8
970
971 !-----------------------------------------------------------
972 ! Point to production rates array in physics buffer where
973 ! rates will be stored for ionosphere module access. Also,
974 ! initialize rates to zero before column loop since only
975 ! daylight values are filled
976 !-----------------------------------------------------------
977 if (indxAIPRS>0) then
978 call pbuf_get_field(pbuf, indxAIPRS, aurIPRateSum)
979 aurIPRateSum(:,:) = 0._r8
980 endif
981
982level_loop : &
983 do k = 1,lev1
984 where( do_aurora(:) )
985 press(:ncol) = 10._r8*pmid(:ncol,k) ! from Pa to dyne/cm^2
986 tempi(:ncol) = tn(:ncol,k)
987 barm(:) = mbar(:,k)
988 p0ez(:) = (press(:)/(grav*4.e-6_r8))**.606_r8
989 xalfa1(:) = p0ez(:)/alfa1(:)
990 xalfa2(:) = p0ez(:)/alfa2(:)
991 xcusp (:) = p0ez(:)/alfac
992 xdrizl(:) = p0ez(:)/alfad
993
994!-----------------------------------------------------------------------
995! ... initialize (whole array operations):
996!-----------------------------------------------------------------------
997 flux1_ion(:) = const0
998 flux2_ion(:) = const0
999 alfa1_ion(:) = const0
1000 alfa2_ion(:) = const0
1001 cusp_ion(:) = const0
1002 drizl_ion(:) = const0
1003 endwhere
1004!-----------------------------------------------------------------------
1005! ... auroral electrons
1006!-----------------------------------------------------------------------
1007 call aion( xalfa1, alfa1_ion, do_aurora, ncol )
1008 call aion( xalfa2, alfa2_ion, do_aurora, ncol )
1009 call aion( xcusp , cusp_ion, do_aurora, ncol )
1010 call aion( xdrizl, drizl_ion, do_aurora, ncol )
1011 where( do_aurora(:) )
1012 falfa1(:) = alfa1(:)*flux1(:) ! s7
1013 falfa2(:) = alfa2(:)*flux2(:) ! s8
1014 fcusp (:) = cusp(:)*alfac*fc ! s9
1015 fdrizl(:) = drizl(:)*alfad*fd ! s10
1016 qsum(:) = falfa1(:)*alfa1_ion(:) & ! s7*s3
1017 + falfa2(:)*alfa2_ion(:) & ! s8*s4
1018 + fcusp(:)*cusp_ion (:) & ! s9*s5
1019 + fdrizl(:)*drizl_ion(:) ! s10*s6
1020 endwhere
1021!-----------------------------------------------------------------------
1022! ... include solar protons if add_sproton is set,
1023! and high energy electrons if add_helectron is set
1024!-----------------------------------------------------------------------
1025!-----------------------------------------------------------------------
1026! ... solar protons
1027!-----------------------------------------------------------------------
1028 if( add_sproton ) then
1029 if( flx_sp > 1.e-19_r8 ) then
1030 where( do_aurora(:) )
1031 p0ez_mbar(:) = press(:)/(boltz_cgs*tempi(:)*barm(:))/avo
1032 tk_mbar(:) = gask*tempi(:)/(grav*barm(:))*p0ez_mbar(:)
1033 xalfa_sp(:) = ((tk_mbar(:)/.00271_r8)**.58140_r8)/alfa_sp
1034 endwhere
1035 call bion( xalfa_sp, alfasp_bion, do_aurora, ncol )
1036 where( do_aurora(:) )
1037 falfa_sp(:) = drizl(:)*p0ez_mbar(:)*flx_sp*1.e6_r8/(tk_mbar(:)*35._r8)
1038 qsum(:) = qsum(:) + falfa_sp(:)*alfasp_bion(:)
1039 endwhere
1040 end if
1041 end if
1042!-----------------------------------------------------------------------
1043! ... high energy electrons
1044!-----------------------------------------------------------------------
1045 if( add_helectron ) then
1046 if( e30 > 1.e-19_r8 ) then
1047 where( do_aurora(:) )
1048 xalfa3(:) = p0ez(:)/alfa3(:) ! alfa3(:)==alfa30
1049 endwhere
1050 call aion( xalfa3, alfa3_ion, do_aurora, ncol )
1051 where( do_aurora(:) )
1052 falfa3(:) = alfa3(:)*flux3(:) ! s13 (high energy electrons)
1053 qsum(:) = qsum(:) + falfa3(:)*alfa3_ion(:)
1054 endwhere
1055 end if
1056 end if
1057
1058!-----------------------------------------------------------------------
1059! ... form production
1060!-----------------------------------------------------------------------
1061 where( do_aurora(:) )
1062 barm_t(:) = grav*barm(:)/(35.e-3_r8*gask*tempi(:))
1063 qsum(:) = qsum(:)*barm_t(:) ! s1 = s1*s11
1064 wrk(:,k) = qsum(:)
1065!-----------------------------------------------------------------------
1066! ... denominator of equations (13-16) in Roble,1987.
1067!-----------------------------------------------------------------------
1068 xn2(:) = max( (1._r8 - o2(:,k) - o1(:,k)),1.e-8_r8 )
1069 denom(:) = 0.92_r8*xn2(:)*rmassinv_n2 &
1070 + 1.5_r8*o2(:,k) *rmassinv_o2 + 0.56_r8*o1(:,k) *rmassinv_o1
1071!-----------------------------------------------------------------------
1072! ... production of O2+ (equation (15) in Roble,1987):
1073!-----------------------------------------------------------------------
1074 qo2p_aur(:) = qsum(:)*o2(:,k)/(rmass_o2*denom(:)) + qia(2)
1075!-----------------------------------------------------------------------
1076! ... production of O+ (equation (16) in Roble,1987):
1077!-----------------------------------------------------------------------
1078 qop_aur(:) = qsum(:)*(.5_r8 *o2(:,k)*rmassinv_o2 &
1079 + .56_r8*o1(:,k)*rmassinv_o1)/denom(:) + qia(3)
1080!-----------------------------------------------------------------------
1081! ... production of N2+ (equation (13) in Roble,1987)
1082!-----------------------------------------------------------------------
1083 qn2p_aur(:) = qsum(:)*.7_r8*xn2(:)/(rmass_n2*denom(:)) + qia(1)
1084 qo2p(:,k) = qo2p(:,k) + qo2p_aur(:)
1085 qop(:,k) = qop(:,k) + qop_aur(:)
1086 qn2p(:,k) = qn2p(:,k) + qn2p_aur(:)
1087 qnp(:,k) = qnp (:,k) + .22_r8/.7_r8 * qn2p_aur(:)
1088 endwhere
1089 end do level_loop
1090
1091 !----------------------------------------------------------------
1092 ! Store the sum of the ion production rates in pbuf to be used
1093 ! in the ionosx module
1094 !----------------------------------------------------------------
1095 if (indxAIPRS>0) then
1096
1097 aurIPRateSum(1:ncol,1:pver) = wrk(1:ncol,1:pver)
1098
1099 endif
1100
1101 call outfld( 'QSUM', wrk, ncol, lchnk )
1102
1103 end subroutine aurora_ions

1106 flux1, flux2, flux3, tn, o2, &
1107 o1, mbar, tpions, pmid, do_aurora, &
1108 ncol, lchnk )
1109!-----------------------------------------------------------------------
1110! ... calculate auroral additions to ionization rates
1111!-----------------------------------------------------------------------
1112
1113 use ppgrid, only : pcols, pver
1114 use cam_history, only : outfld
1115
1116 implicit none
1117
