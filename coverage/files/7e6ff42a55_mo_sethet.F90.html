853 het_rates(i,k,hcooh_ndx) = max( rain(i,k) / (h2o_mol*(work1(i) + 1._r8/(xhen_ch3cooh(i,k)*work2(i)))),0._r8 )
854 endif
855 if ( hcn_ndx > 0 ) then
856 het_rates(i,k,hcn_ndx ) = max( rain(i,k) / (h2o_mol*(work1(i) + 1._r8/(xhen_hcn(i,k)*work2(i)))),0._r8 )
857 endif
858 if ( ch3cn_ndx > 0 ) then
859 het_rates(i,k,ch3cn_ndx ) = max( rain(i,k) / (h2o_mol*(work1(i) + 1._r8/(xhen_ch3cn(i,k)*work2(i)))),0._r8 )
860 endif
861
862 end do Column_loop2
863 end do level_loop2
864
865 !-----------------------------------------------------------------
866 ! ... Set rates above tropopause = 0.
867 !-----------------------------------------------------------------
868 do mm = 1,gas_wetdep_cnt
869 m = wetdep_map(mm)
870 do i = 1,ncol
871 do k = 1,ktop(i)
872 het_rates(i,k,m) = 0._r8
873 end do
874 end do
875 if ( any( het_rates(:ncol,:,m) == MISSING) ) then
876 write(hetratestrg,'(I3)') m
877 call endrun('sethet: het_rates (wet dep) not set for het reaction number : '//hetratestrg)
878 endif
879 end do

881 end subroutine sethet
