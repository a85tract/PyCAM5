851subroutine prim_energy_halftimes(elem,hvcoord,tl,n,t_before_advance,nets,nete)
852!
853! called at the end of a timestep, before timelevel update. Solution known at
854! dynamics: nm1, n0, np1.
855!
856! LF code RK code
857! This routine is called 4 times: n=1: t1=nm1, t2=n0 start after forcing, before timestep
858! n=2: t1=n0, t2=np1 after timestep after timestep, including remap
859! n=3: t1=n0, t2=np1 after forcing after fixer
860! n=4: t1=n0, t2=np1 after Robert before forcing
861!
862! LF case we use staggered formulas:
863! compute the energies at time half way between timelevels t1 and t2,
864! using timelevels t1,t2. store in location n of our elem()%accum%* variables
865!
866! in the Dry case, PE (and similar for IE) will be conserved exactly by Leapfrog,
867! if it is defined as:
868!
869! PE(n+.5) = .5*( T(n+1) dp(n) + T(n) dp(n+1) )
870!
871! KE cannot be exactly conserved with Leapfrog. but we still use:
872!
873! KE(n+.5) = .5*( .5 u(n+1)^2 dp(n) + .5 u(n)^2 dp(n+1) )
874!
875!
876 use kinds, only : real_kind
877 use dimensions_mod, only : np, np, nlev
878 use control_mod, only : use_cpstar
879 use hybvcoord_mod, only : hvcoord_t
880 use element_mod, only : element_t
881 use physical_constants, only : Cp, cpwater_vapor
882 use physics_mod, only : Virtual_Specific_Heat, Virtual_Temperature
883 use prim_si_mod, only : preq_hydrostatic
884
885 integer :: t1,t2,n,nets,nete
886 type (element_t) , intent(inout), target :: elem(:)
887 type (hvcoord_t) :: hvcoord
888 type (TimeLevel_t), intent(in) :: tl
889 logical :: t_before_advance
890
891 integer :: ie,k,i,j,nm_f
892 real (kind=real_kind), dimension(np,np,nlev) :: dpt1,dpt2 ! delta pressure
893 real (kind=real_kind), dimension(np,np) :: E
894 real (kind=real_kind), dimension(np,np) :: suml,suml2,v1,v2
895 real (kind=real_kind), dimension(np,np,nlev) :: sumlk, suml2k
896 real (kind=real_kind), dimension(np,np,nlev) :: p,T_v,phi
897 real (kind=real_kind) :: cp_star1,cp_star2,qval_t1,qval_t2
898 real (kind=real_kind) :: Qt
899 logical :: wet
900
901
902 logical tstagger
903 integer:: t2_qdp, t1_qdp ! the time pointers for Qdp are not the same
904
905 nm_f = 1
906 if (t_before_advance) then
907 t1=tl%nm1
908 t2=tl%n0
909 call TimeLevel_Qdp( tl, qsplit, t2_qdp, t1_qdp) !get n0 level into t2_qdp
910 else
911 t1=tl%n0
912 t2=tl%np1
913 call TimeLevel_Qdp(tl, qsplit, t1_qdp, t2_qdp) !get np1 into t2_qdp
914 endif
915
916
917
918! energy_fixer
919! <0 disabled, but compute energy non-staggered in time
920! 0 disabled, but compute energy staggered in time (for use with leapfrog code)
921! >0 Enabled. energy fixer requires dry formulation (use_cpstar=0) and non-staggered in time
922!
923 tstagger = .false.
924 if (energy_fixer==0) tstagger = .true.
925
926
927 ! IE Cp*dpdn*T + (Cpv-Cp) Qdpdn*T
928 ! Cp*dpdn(n)*T(n+1) + (Cpv-Cp) Qdpdn(n)*T(n+1)
929 ! [Cp + (Cpv-Cp) Q(n)] *dpdn(n)*T(n+1)
930 do ie=nets,nete
931
932#if (defined COLUMN_OPENMP)
933!$omp parallel do private(k)
934#endif
935 do k=1,nlev
936 dpt1(:,:,k) = ( hvcoord%hyai(k+1) - hvcoord%hyai(k) )*hvcoord%ps0 + &
937 ( hvcoord%hybi(k+1) - hvcoord%hybi(k) )*elem(ie)%state%ps_v(:,:,t1)
938 dpt2(:,:,k) = ( hvcoord%hyai(k+1) - hvcoord%hyai(k) )*hvcoord%ps0 + &
939 ( hvcoord%hybi(k+1) - hvcoord%hybi(k) )*elem(ie)%state%ps_v(:,:,t2)
940 enddo
941
942#if (defined COLUMN_OPENMP)
943!$omp parallel do private(k,i,j,cp_star1,cp_star2,qval_t1,qval_t2)
944#endif
945 do k=1,nlev
946 sumlk(:,:,k)=0
947 suml2k(:,:,k)=0

949 do j=1,np
950 if(use_cpstar == 1) then
951 ! Cp_star = cp + (Cpwater_vapor - cp)*qval
952 qval_t1 = elem(ie)%state%Qdp(i,j,k,1,t1_qdp)/dpt1(i,j,k)

955 cp_star2= Virtual_Specific_Heat(qval_t2)

962 sumlk(i,j,k) = sumlk(i,j,k) + Cp_star2*elem(ie)%state%T(i,j,k,t1) *dpt2(i,j,k)/2
963 suml2k(i,j,k) = suml2k(i,j,k) + (cp_star1-cp)*elem(ie)%state%T(i,j,k,t2) *dpt1(i,j,k)/2
964 suml2k(i,j,k) = suml2k(i,j,k) + (cp_star2-cp)*elem(ie)%state%T(i,j,k,t1) *dpt2(i,j,k)/2
965 else
966 sumlk(i,j,k) = sumlk(i,j,k) + Cp_star2*elem(ie)%state%T(i,j,k,t2) *dpt2(i,j,k)
967 suml2k(i,j,k) = suml2k(i,j,k) + (cp_star2-cp)*elem(ie)%state%T(i,j,k,t2) *dpt2(i,j,k)
968 endif
969 enddo
970 enddo
971 enddo
972 suml=0
973 suml2=0
974 do k=1,nlev
975 suml(:,:) = suml(:,:) + sumlk(:,:,k)
976 suml2(:,:) = suml2(:,:) + suml2k(:,:,k)
977 enddo
978 elem(ie)%accum%IEner(:,:,n)=suml(:,:)
979 elem(ie)%accum%IEner_wet(:,:,n)=suml2(:,:)
980
981
982 ! KE .5 dp/dn U^2
983#if (defined COLUMN_OPENMP)
984!$omp parallel do private(k,E)
985#endif
986 do k=1,nlev
987 if (tstagger) then
988 E = (elem(ie)%state%v(:,:,1,k,t2)**2 + &
989 elem(ie)%state%v(:,:,2,k,t2)**2 ) / 2
990 sumlk(:,:,k) = E*dpt1(:,:,k)/2
991 E = (elem(ie)%state%v(:,:,1,k,t1)**2 + &
992 elem(ie)%state%v(:,:,2,k,t1)**2 ) / 2
993 sumlk(:,:,k) = sumlk(:,:,k) + E*dpt2(:,:,k)/2
994 else
995 E = (elem(ie)%state%v(:,:,1,k,t2)**2 + &
996 elem(ie)%state%v(:,:,2,k,t2)**2 ) / 2
997 sumlk(:,:,k) = E*dpt2(:,:,k)
998 endif
999 enddo
1000 suml=0
1001 do k=1,nlev
1002 suml(:,:) = suml(:,:) + sumlk(:,:,k)
1003 enddo
1004 elem(ie)%accum%KEner(:,:,n)=suml(:,:)
1005
1006
1007
1008 ! PE dp/dn PHIs
1009 suml=0
1010 do k=1,nlev
1011 if (tstagger) then
1012 suml = suml + elem(ie)%state%phis(:,:)*dpt1(:,:,k)/2
1013 suml = suml + elem(ie)%state%phis(:,:)*dpt2(:,:,k)/2
1014 else
1015 suml = suml + elem(ie)%state%phis(:,:)*dpt2(:,:,k)
1016 endif
1017 enddo
1018 elem(ie)%accum%PEner(:,:,n)=suml(:,:)
1019
1020
1021
1022! compute alternate PE term which matches what is used in CAM physics
1023 wet =(moisture /= "dry")
1024 do k=1,nlev
1025 p(:,:,k) = hvcoord%hyam(k)*hvcoord%ps0 + hvcoord%hybm(k)*elem(ie)%state%ps_v(:,:,t2)
1026 do j=1,np
1027 do i=1,np
1028 if (wet) then
1029 Qt = elem(ie)%state%Qdp(i,j,k,1,t2_qdp)/dpt2(i,j,k)
1030 T_v(i,j,k) = Virtual_Temperature(elem(ie)%state%T(i,j,k,t2),Qt)
1031 else
1032 T_v(i,j,k) = elem(ie)%state%T(i,j,k,t2)
1033 endif
1034 end do
1035 end do
1036 end do
1037
1038 enddo
1039
1040end subroutine prim_energy_halftimes

1045subroutine prim_diag_scalars(elem,hvcoord,tl,n,t_before_advance,nets,nete)

1047! called at the end of a timestep, before timelevel update. Solution known at
1048! timelevel nm1,n0,np1.

1050! This routine is called twice: n=1: t1=nm1, t2=n0
1051! n=2: t1=n0, t2=np1
1052!
1053! in all cases:
1054! Q1mass(n) = Qdp(t2) (or: Q(t2)*dp(t2) )
1055!
1056! We also compute
1057! Qmass(n) = .5*( Q(t2) dp(t1) Q(t1) dp(t2) ) Leapfrog, concentration formulation
1058! Qmass(n) = .5*( Qdp(t1) + Qdp(t2) ) Leapfrog, conservation formulation
1059! Qmass(n) = Qdp(t2) RK2, conservation formulation
1060!
1061!

1064 use hybvcoord_mod, only : hvcoord_t
1065 use element_mod, only : element_t

1069 type (hvcoord_t) :: hvcoord
1070
1071 integer :: ie,k,q,nm_f
1072 real (kind=real_kind), dimension(np,np) :: ps ! pressure
1073 real (kind=real_kind), dimension(np,np) :: dp ! delta pressure
1074 real (kind=real_kind), dimension(np,np) :: E
1075 real (kind=real_kind), dimension(np,np) :: suml

1079
1080 nm_f = 1
1081 if (t_before_advance) then
1082 t1=tl%nm1
1083 t2=tl%n0
1084 call TimeLevel_Qdp( tl, qsplit, t2_qdp) !get n0 level into t2_qdp
1085 else
1086 t1=tl%n0
1087 t2=tl%np1
1088 call TimeLevel_Qdp(tl, qsplit, tmp, t2_qdp) !get np1 into t2_qdp (don't need tmp)
1089 endif
1090
1091
1092 !
1093 ! RK2 forward scheme. compute everything at t2
1094 ! (used by CAM)
1095 ! Q has only one time dimension
1096 if (tstep_type>0) then
1097

1099#if (defined COLUMN_OPENMP)

1106 enddo

1108 enddo
1109 enddo
1110
1111 do ie=nets,nete
1112#if (defined COLUMN_OPENMP)
1113 !$omp parallel do private(q,k,suml)
1114#endif
1115 do q=1,qsize
1116 suml=0
1117 do k=1,nlev
1118 suml = suml + elem(ie)%state%Qdp(:,:,k,q,t2_qdp)
1119 enddo
1120 elem(ie)%accum%Q1mass(:,:,q)=suml(:,:)
1121 elem(ie)%accum%Qmass(:,:,q,n)=suml(:,:)
1122 enddo
1123 enddo
1124 endif
