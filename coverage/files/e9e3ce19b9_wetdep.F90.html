915 if (dblchek(i) .lt. 0._r8) then
916 write(iulog,*) ' wetdapa: negative value ', i, k, tracer(i,k), &
917 dblchek(i), scavt(i,k), srct(i), rat(i), fracev(i)
918 endif
919 end do
920 endif
921
922 end do
923
924 end subroutine wetdepa_v1

926!==============================================================================
927
928! wetdepg is currently being used for both CAM4 and CAM5 by making use of the
929! cam_physpkg_is method.
930
931 subroutine wetdepg( t, p, q, pdel, &
932 cldt, cldc, cmfdqr, evapc, precs, evaps, &
933 rain, cwat, tracer, deltat, molwt, &
934 solconst, scavt, iscavt, cldv, icwmr1, &
935 icwmr2, fracis, ncol )
936
937 !-----------------------------------------------------------------------
938 ! Purpose:
939 ! scavenging of gas phase constituents by henry's law
940 !
941 ! Author: P. Rasch
942 !-----------------------------------------------------------------------
943
944 real(r8), intent(in) ::&
945 t(pcols,pver), &! temperature
946 p(pcols,pver), &! pressure
947 q(pcols,pver), &! moisture
948 pdel(pcols,pver), &! pressure thikness
949 cldt(pcols,pver), &! total cloud fraction
950 cldc(pcols,pver), &! convective cloud fraction
951 cmfdqr(pcols,pver), &! rate of production of convective precip
952 rain (pcols,pver), &! total rainwater mixing ratio
953 cwat(pcols,pver), &! cloud water amount
954 precs(pcols,pver), &! rate of production of stratiform precip
955 evaps(pcols,pver), &! rate of evaporation of precip
956! Sungsu
957 evapc(pcols,pver), &! Rate of evaporation of convective precipitation
958! Sungsu
959 cldv(pcols,pver), &! estimate of local volume occupied by clouds
960 icwmr1 (pcols,pver), &! in cloud water mixing ration for zhang scheme
961 icwmr2 (pcols,pver), &! in cloud water mixing ration for hack scheme
962 deltat, &! time step
963 tracer(pcols,pver), &! trace species
964 molwt ! molecular weights
965
966 integer, intent(in) :: ncol
967
968 real(r8) &
969 solconst(pcols,pver) ! Henry's law coefficient
970
971 real(r8), intent(out) ::&
972 scavt(pcols,pver), &! scavenging tend
973 iscavt(pcols,pver), &! incloud scavenging tends
974 fracis(pcols, pver) ! fraction of constituent that is insoluble
975
976 ! local variables
977
978 integer i ! x index
979 integer k ! z index
980
981 real(r8) adjfac ! factor stolen from cmfmca
982 real(r8) aqfrac ! fraction of tracer in aqueous phase
983 real(r8) cwatc ! local convective total water amount
984 real(r8) cwats ! local stratiform total water amount
985 real(r8) cwatl ! local cloud liq water amount
986 real(r8) cwatp ! local water amount falling from above precip
987 real(r8) cwatpl ! local water amount falling from above precip (liq)
988 real(r8) cwatt ! local sum of strat + conv total water amount
989 real(r8) cwatti ! cwatt/cldv = cloudy grid volume mixing ratio
990 real(r8) fracev ! fraction of precip from above that is evaporating
991 real(r8) fracp ! fraction of cloud water converted to precip
992 real(r8) gafrac ! fraction of tracer in gas phasea
993 real(r8) hconst ! henry's law solubility constant when equation is expressed
994 ! in terms of mixing ratios
995 real(r8) mpla ! moles / liter H2O entering the layer from above
996 real(r8) mplb ! moles / liter H2O leaving the layer below
997 real(r8) omsm ! 1 - (a small number)
998 real(r8) part ! partial pressure of tracer in atmospheres
999 real(r8) patm ! total pressure in atmospheres
1000 real(r8) pdog ! work variable (pdel/gravit)
1001 real(r8) precab(pcols) ! precip from above (work array)
1002 real(r8) precbl ! precip work variable
1003 real(r8) precxx ! precip work variable
1004 real(r8) precxx2 !
1005 real(r8) precic ! precip work variable
1006 real(r8) rat ! ratio of amount available to amount removed
1007 real(r8) scavab(pcols) ! scavenged tracer flux from above (work array)
1008 real(r8) scavabc(pcols) ! scavenged tracer flux from above (work array)
1009
1010 real(r8) scavmax ! an estimate of the max tracer avail for removal
1011 real(r8) scavbl ! flux removed at bottom of layer
1012 real(r8) fins ! in cloud fraction removed by strat rain
1013 real(r8) finc ! in cloud fraction removed by conv rain
1014 real(r8) rate ! max removal rate estimate
1015 real(r8) scavlimt ! limiting value 1
1016 real(r8) scavt1 ! limiting value 2
1017 real(r8) scavin ! scavenging by incloud processes
1018 real(r8) scavbc ! scavenging by below cloud processes
1019 real(r8) tc
1020 real(r8) weight ! ice fraction
1021 real(r8) wtpl ! work variable
1022 real(r8) cldmabs(pcols) ! maximum cloud at or above this level
1023 real(r8) cldmabc(pcols) ! maximum cloud at or above this level
1024 !-----------------------------------------------------------
1025
1026 omsm = 1._r8-2*epsilon(1._r8) ! used to prevent roundoff errors below zero
1027
1028 adjfac = deltat/(max(deltat,cmftau)) ! adjustment factor from hack scheme
1029
1030 ! zero accumulators
1031 do i = 1,pcols
1032 precab(i) = 1.e-36_r8
1033 scavab(i) = 0._r8
1034 cldmabs(i) = 0._r8
1035 end do
1036
1037 do k = 1,pver
1038 do i = 1,ncol
1039
1040 tc = t(i,k) - tmelt
1041 weight = max(0._r8,min(-tc*0.05_r8,1.0_r8)) ! fraction of condensate that is ice
1042
1043 cldmabs(i) = max(cldmabs(i),cldt(i,k))
1044
1045 ! partitioning coefs for gas and aqueous phase
1046 ! take as a cloud water amount, the sum of the stratiform amount
1047 ! plus the convective rain water amount
1048
1049 ! convective amnt is just the local precip rate from the hack scheme
1050 ! since there is no storage of water, this ignores that falling from above
1051 ! cwatc = cmfdqr(i,k)*deltat/adjfac
1052 !++mcb -- test cwatc
1053 cwatc = (icwmr1(i,k) + icwmr2(i,k)) * (1._r8-weight)
1054 !--mcb
1055
1056 ! strat cloud water amount and also ignore the part falling from above
1057 cwats = cwat(i,k)
1058
1059 ! cloud water as liq
1060 !++mcb -- add cwatc later (in cwatti)
1061 ! cwatl = (1.-weight)*(cwatc+cwats)
1062 cwatl = (1._r8-weight)*cwats
1063 ! cloud water as ice
1064 !*not used cwati = weight*(cwatc+cwats)
1065
1066 ! total suspended condensate as liquid
1067 cwatt = cwatl + rain(i,k)
1068
1069 ! incloud version
1070 !++mcb -- add cwatc here
1071 cwatti = cwatt/max(cldv(i,k), 0.00001_r8) + cwatc
1072
1073 ! partitioning terms
1074 patm = p(i,k)/1.013e5_r8 ! pressure in atmospheres
1075 hconst = molwta*patm*solconst(i,k)*cwatti/rhoh2o
1076 aqfrac = hconst/(1._r8+hconst)
1077 gafrac = 1/(1._r8+hconst)
1078 fracis(i,k) = gafrac
1079
1080
1081 ! partial pressure of the tracer in the gridbox in atmospheres
1082 part = patm*gafrac*tracer(i,k)*molwta/molwt
1083
1084 ! use henrys law to give moles tracer /liter of water
1085 ! in this volume
1086 ! then convert to kg tracer /liter of water (kg tracer / kg water)
1087 mplb = solconst(i,k)*part*molwt/1000._r8
1088
1089
1090 pdog = pdel(i,k)/gravit
1091
1092 ! this part of precip will be carried downward but at a new molarity of mpl
1093 precic = pdog*(precs(i,k) + cmfdqr(i,k))
1094
1095 ! we cant take out more than entered, plus that available in the cloud
1096 ! scavmax = scavab(i)+tracer(i,k)*cldt(i,k)/deltat*pdog
1097 scavmax = scavab(i)+tracer(i,k)*cldv(i,k)/deltat*pdog
1098
1099 ! flux of tracer by incloud processes
1100 scavin = precic*(1._r8-weight)*mplb
1101
1102 ! fraction of precip which entered above that leaves below
1103 if (cam_physpkg_is('cam5')) then
1104 ! Sungsu added evaporation of convective precipitation below.
1105 precxx = precab(i)-pdog*(evaps(i,k)+evapc(i,k))
1106 else
1107 precxx = precab(i)-pdog*evaps(i,k)
1108 end if
1109 precxx = max (precxx,0.0_r8)
1110
1111 ! flux of tracer by below cloud processes
1112 !++mcb -- removed wtpl because it is now not assigned and previously
1113 ! when it was assigned it was unnecessary: if(tc.gt.0)wtpl=1
1114 if (tc.gt.0) then
1115 ! scavbc = precxx*wtpl*mplb ! if liquid
1116 scavbc = precxx*mplb ! if liquid
1117 else
1118 precxx2=max(precxx,1.e-36_r8)
1119 scavbc = scavab(i)*precxx2/(precab(i)) ! if ice
1120 endif
1121
1122 scavbl = min(scavbc + scavin, scavmax)
1123
1124 ! first guess assuming that henries law works
1125 scavt1 = (scavab(i)-scavbl)/pdog*omsm
1126
1127 ! pjr this should not be required, but we put it in to make sure we cant remove too much
1128 ! remember, scavt1 is generally negative (indicating removal)
1129 scavt1 = max(scavt1,-tracer(i,k)*cldv(i,k)/deltat)
1130
1131 !++mcb -- remove this limitation for gas species
1132 !c use the dana and hales or balkanski limit on scavenging
1133 !c rate = precab(i)*0.1
1134 ! rate = (precic + precxx)*0.1
1135 ! scavlimt = -tracer(i,k)*cldv(i,k)
1136 ! $ *rate/(1.+rate*deltat)
1137
1138 ! scavt(i,k) = max(scavt1, scavlimt)
1139
1140 ! instead just set scavt to scavt1
1141 scavt(i,k) = scavt1
1142 !--mcb
1143
1144 ! now update the amount leaving the layer
1145 scavbl = scavab(i) - scavt(i,k)*pdog
1146
1147 ! in cloud amount is that formed locally over the total flux out bottom
1148 fins = scavin/(scavin + scavbc + 1.e-36_r8)
1149 iscavt(i,k) = scavt(i,k)*fins
1150
1151 scavab(i) = scavbl
1152 precab(i) = max(precxx + precic,1.e-36_r8)
1153
1154
1155
1156 end do
1157 end do
1158
1159 end subroutine wetdepg
