535 call endrun('chem_surfvals_set_all: incorrect ghg_yearStart_model')
536 endif
537 yrmodel = yearRan_model + ghg_yearStart_data
538
539 nyrm = ghg_yearStart_data + yearRan_model - yrdata(1) + 1
540 end if
541 end if
542
543 nyrp = nyrm + 1
544!
545! if current date is before yrdata(1), quit
546!
547 if (nyrm < 1) then
548 write(iulog,*)'chem_surfvals_set_all: data time index is out of bounds'
549 write(iulog,*)'nyrm = ',nyrm,' nyrp= ',nyrp, ' ncdate= ', ncdate
550 call endrun
551 endif
552!
553! if current date later than yrdata(ntim), call endrun.
554! if want to use ntim values - uncomment the following lines
555! below and comment the call to endrun and previous write
556!
557 if (nyrp > ntim) then
558 call endrun ('chem_surfvals_set_all: error - current date is past the end of valid data')
559! write(iulog,*)'chem_surfvals_set_all: using ghg data for ',yrdata(ntim)
560! co2vmr = co2(ntim)*1.e-06
561! ch4vmr = ch4(ntim)*1.e-09
562! n2ovmr = n2o(ntim)*1.e-09
563! f11vmr = f11(ntim)*1.e-12*(1.+cfcscl)
564! f12vmr = f12(ntim)*1.e-12
565! co2mmr = rmwco2 * co2vmr
566! return
567 endif
568!
569! determine time interpolation factors, check sanity
570! of interpolation factors to within 32-bit roundoff
571! assume that day of year is 1 for all input data
572!
573 doymodel = yrmodel*365._r8 + calday
574 doydatam = yrdata(nyrm)*365._r8 + 1._r8
575 doydatap = yrdata(nyrp)*365._r8 + 1._r8
576
577 call get_timeinterp_factors(.false.,2,doydatam,doydatap, doymodel, &
578 fact1, fact2,'chem_surfvals')
579
580!
581! do time interpolation:
582! co2 in ppmv
583! n2o,ch4 in ppbv
584! f11,f12 in pptv
585!
586 co2vmr = (co2(nyrm)*fact1 + co2(nyrp)*fact2)*1.e-06_r8
587 ch4vmr = (ch4(nyrm)*fact1 + ch4(nyrp)*fact2)*1.e-09_r8
588 n2ovmr = (n2o(nyrm)*fact1 + n2o(nyrp)*fact2)*1.e-09_r8
589
590 cfcscl = (adj(nyrm)*fact1 + adj(nyrp)*fact2)
591 f11vmr = (f11(nyrm)*fact1 + f11(nyrp)*fact2)*1.e-12_r8*(1._r8+cfcscl)
592 f12vmr = (f12(nyrm)*fact1 + f12(nyrp)*fact2)*1.e-12_r8
593
594 return
595end subroutine chem_surfvals_set_all

599subroutine chem_surfvals_set_co2()
600!-----------------------------------------------------------------------
601!
602! Purpose:
603! Computes co2 greenhouse gas volume mixing ratio via ramping info
604! provided in namelist var's
605!
606! Author: B. Eaton - updated ramp_ghg for use in chem_surfvals module
607!
608!-----------------------------------------------------------------------
609 use shr_kind_mod, only: r8 => shr_kind_r8
610
611!---------------------------Local variables-----------------------------
612
613 real(r8) :: daydiff ! number of days of co2 ramping
614 integer :: yr, mon, day, ncsec ! components of a date
615 integer :: ncdate ! current date in integer format [yyyymmdd]
616!-----------------------------------------------------------------------
617
618 call get_curr_date(yr, mon, day, ncsec)
619 ncdate = yr*10000 + mon*100 + day
620
621 call timemgr_datediff(co2_start, 0, ncdate, ncsec, daydiff)
622
623 if (daydiff > 0.0_r8) then
624
625 co2vmr = co2_base*(co2_daily_factor)**daydiff
626
627 if(co2_daily_factor < 1.0_r8) then
628 co2vmr = max(co2vmr,co2_limit)
629 else
630 co2vmr = min(co2vmr,co2_limit)
631 end if
632 end if
633
634 return
635end subroutine chem_surfvals_set_co2
