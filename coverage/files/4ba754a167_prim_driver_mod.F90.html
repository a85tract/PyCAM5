1767 subroutine prim_energy_fixer(elem,hvcoord,hybrid,tl,nets,nete,nsubstep)
1768!
1769! non-subcycle code:
1770! Solution is given at times u(t-1),u(t),u(t+1)
1771! E(n=1) = energy before dynamics
1772! E(n=2) = energy after dynamics
1773!
1774! fixer will add a constant to the temperature so E(n=2) = E(n=1)
1775!
1776 use parallel_mod, only: global_shared_buf, global_shared_sum
1777 use kinds, only : real_kind
1778 use hybvcoord_mod, only : hvcoord_t
1779 use physical_constants, only : Cp
1780 use time_mod, only : timelevel_t
1781 use control_mod, only : use_cpstar, energy_fixer
1782 use hybvcoord_mod, only : hvcoord_t
1783 use global_norms_mod, only: wrap_repro_sum
1784 use parallel_mod, only : abortmp
1785 type (hybrid_t), intent(in) :: hybrid ! distributed parallel structure (shared)
1786 integer :: t2,n,nets,nete
1787 type (element_t) , intent(inout), target :: elem(:)
1788 type (hvcoord_t) :: hvcoord
1789 type (TimeLevel_t), intent(inout) :: tl
1790 integer, intent(in) :: nsubstep
1791
1792 integer :: ie,k,i,j,nmax
1793 real (kind=real_kind), dimension(np,np,nlev) :: dp ! delta pressure
1794 real (kind=real_kind), dimension(np,np,nlev) :: sumlk
1795 real (kind=real_kind), pointer :: PEner(:,:,:)
1796 real (kind=real_kind), dimension(np,np) :: suml
1797 real (kind=real_kind) :: psum(nets:nete,4),psum_g(4),beta
1798
1799 ! when forcing is applied during dynamics timstep, actual forcing is
1800 ! slightly different (about 0.1 W/m^2) then expected by the physics
1801 ! since u & T are changing while FU and FT are held constant.
1802 ! to correct for this, save compute de_from_forcing at step 1
1803 ! and then adjust by: de_from_forcing_step1 - de_from_forcing_stepN
1804 real (kind=real_kind),save :: de_from_forcing_step1
1805 real (kind=real_kind) :: de_from_forcing
1806
1807 t2=tl%np1 ! timelevel for T
1808 if (use_cpstar /= 0 ) then
1809 call abortmp('Energy fixer requires use_cpstar=0')
1810 endif
1811
1812
1813 psum = 0
1814 do ie=nets,nete
1815
1816#if (defined COLUMN_OPENMP)
1817!$omp parallel do default(shared), private(k)
1818#endif
1819 do k=1,nlev
1820 dp(:,:,k) = ( hvcoord%hyai(k+1) - hvcoord%hyai(k) )*hvcoord%ps0 + &
1821 ( hvcoord%hybi(k+1) - hvcoord%hybi(k) )*elem(ie)%state%ps_v(:,:,t2)
1822 enddo
1823 suml=0
1824#if (defined COLUMN_OPENMP)
1825!$omp parallel do default(shared), private(k, i, j)
1826#endif
1827 do k=1,nlev
1828 do i=1,np
1829 do j=1,np
1830 sumlk(i,j,k) = cp*dp(i,j,k)
1831 enddo
1832 enddo
1833 enddo
1834 suml=0
1835 do k=1,nlev
1836 do i=1,np
1837 do j=1,np
1838 suml(i,j) = suml(i,j) + sumlk(i,j,k)
1839 enddo
1840 enddo
1841 enddo
1842 PEner => elem(ie)%accum%PEner(:,:,:)
1843
1844 ! psum(:,4) = energy before forcing
1845 ! psum(:,1) = energy after forcing, before dynamics
1846 ! psum(:,2) = energy after dynamics
1847 ! psum(:,3) = cp*dp (internal energy added is beta*psum(:,3))
1848 psum(ie,3) = psum(ie,3) + SUM(suml(:,:)*elem(ie)%spheremp(:,:))
1849 do n=1,2
1850 psum(ie,n) = psum(ie,n) + SUM( elem(ie)%spheremp(:,:)*&
1851 (PEner(:,:,n) + &
1852 elem(ie)%accum%IEner(:,:,n) + &
1853 elem(ie)%accum%KEner(:,:,n) ) )
1854 enddo
1855 enddo
1856
1857 nmax=3
1858
1859 do ie=nets,nete
1860 do n=1,nmax
1861 global_shared_buf(ie,n) = psum(ie,n)
1862 enddo
1863 enddo
1864 call wrap_repro_sum(nvars=nmax, comm=hybrid%par%comm)
1865 do n=1,nmax
1866 psum_g(n) = global_shared_sum(n)
1867 enddo
1868
1869 beta = ( psum_g(1)-psum_g(2) )/psum_g(3)
1870
1871 ! apply fixer
1872 do ie=nets,nete
1873 elem(ie)%state%T(:,:,:,t2) = elem(ie)%state%T(:,:,:,t2) + beta
1874 enddo
1875 end subroutine prim_energy_fixer
