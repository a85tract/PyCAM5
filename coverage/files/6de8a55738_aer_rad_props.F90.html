494 - wrh(icol,ilev) * ssa(krh(icol,ilev), iswband)

499 tau_w (icol, ilev, iswband) = mass(icol, ilev) * ext1 * ssa1
500 tau_w_g(icol, ilev, iswband) = mass(icol, ilev) * ext1 * ssa1 * asm1
501 tau_w_f(icol, ilev, iswband) = mass(icol, ilev) * ext1 * ssa1 * asm1 * asm1

544subroutine get_volcanic_radius_rad_props(lchnk, ncol, mass, pbuf, r_ext, r_scat, r_ascat, r_mu, &
545 tau, tau_w, tau_w_g, tau_w_f)
546
547
548 use physics_buffer, only : pbuf_get_field, pbuf_get_index
549
550 ! Arguments
551 integer, intent(in) :: lchnk
552 integer, intent(in) :: ncol
553 real(r8), intent(in) :: mass(pcols, pver)
554
555 type(physics_buffer_desc), pointer :: pbuf(:)
556 real(r8), intent(in) :: r_ext(:,:)
557 real(r8), intent(in) :: r_scat(:,:)
558 real(r8), intent(in) :: r_ascat(:,:)
559 real(r8), intent(in) :: r_mu(:) ! log(radius) domain of mass-specific optics
560
561 real(r8), intent(out) :: tau (pcols, pver, nswbands)
562 real(r8), intent(out) :: tau_w (pcols, pver, nswbands)
563 real(r8), intent(out) :: tau_w_g(pcols, pver, nswbands)
564 real(r8), intent(out) :: tau_w_f(pcols, pver, nswbands)
565
566 ! Local variables
567 integer :: iswband
568 real(r8) :: g
569
570 integer :: idx ! index to radius in physics buffer
571 real(r8), pointer :: geometric_radius(:,:) ! geometric mean radius of volcanic aerosol
572 real(r8) :: mu(pcols,pver) ! log(geometric mean radius of volcanic aerosol)
573 integer :: imu ! index into table values of log radius
574 integer :: kmu, nmu
575 real(r8) :: wmu, mutrunc, r_mu_max, r_mu_min
576
577 ! interpolated values from table
578 real(r8) :: ext(nswbands)
579 real(r8) :: scat(nswbands)
580 real(r8) :: ascat(nswbands)
581
582 integer :: i, k ! column level iterator
583 !-----------------------------------------------------------------------------
584
585 tau =0._r8
586 tau_w =0._r8
587 tau_w_g=0._r8
588 tau_w_f=0._r8
589
590 ! get microphysical properties for volcanic aerosols
591 idx = pbuf_get_index('VOLC_RAD_GEOM')
592 call pbuf_get_field(pbuf, idx, geometric_radius )
593
594 ! interpolate in radius
595 ! caution: clip the table with no warning when outside bounds
596 nmu = size(r_mu)
597 r_mu_max = r_mu(nmu)
598 r_mu_min = r_mu(1)
599 do i = 1, ncol
600 do k = 1, pver
601 if(geometric_radius(i,k) > 0._r8) then
602 mu(i,k) = log(geometric_radius(i,k))
603 else
604 mu(i,k) = 0._r8
605 endif
606 mutrunc = max(min(mu(i,k),r_mu_max),r_mu_min)
607 kmu = max(min(1 + (mutrunc-r_mu_min)/(r_mu_max-r_mu_min)*(nmu-1),nmu-1._r8),1._r8)
608 wmu = max(min( (mutrunc -r_mu(kmu)) / (r_mu(kmu+1) - r_mu(kmu)) ,1._r8),0._r8)
609 do iswband = 1, nswbands
610 ext(iswband) = &
611 ((1._r8 - wmu) * r_ext(iswband, kmu ) + &
612 (wmu) * r_ext(iswband, kmu+1))
613 scat(iswband) = &
614 ((1._r8 - wmu) * r_scat(iswband, kmu ) + &
615 (wmu) * r_scat(iswband, kmu+1))
616 ascat(iswband) = &
617 ((1._r8 - wmu) * r_ascat(iswband, kmu ) + &
618 (wmu) * r_ascat(iswband, kmu+1))
619 if (scat(iswband).gt.0._r8) then
620 g = ascat(iswband)/scat(iswband)
621 else
622 g=0._r8
623 endif
624 tau (i,k,iswband) = mass(i,k) * ext(iswband)
625 tau_w (i,k,iswband) = mass(i,k) * scat(iswband)
626 tau_w_g(i,k,iswband) = mass(i,k) * ascat(iswband)
627 tau_w_f(i,k,iswband) = mass(i,k) * g * ascat(iswband)
628 end do
629 enddo
630 enddo
631
632end subroutine get_volcanic_radius_rad_props
