148 subroutine ubc_init()
149!-----------------------------------------------------------------------
150! Initialization of time independent fields for the upper boundary condition
151! Calls initialization routine for MSIS, TGCM and SNOE
152!-----------------------------------------------------------------------
153 use mo_tgcm_ubc, only: tgcm_ubc_inti
154 use mo_snoe, only: snoe_inti
155 use mo_msis_ubc, only: msis_ubc_inti
156 use time_manager,only: get_step_size
157 use constituents,only: cnst_get_ind, cnst_mw !Needed for ubc_flux
158 use physics_buffer, only : physics_buffer_desc
159
160!---------------------------Local workspace-----------------------------
161 integer :: steps_per_day
162 logical :: zonal_avg
163!-----------------------------------------------------------------------
164 apply_upper_bc = ptop_ref<1._r8 ! Pa
165
166 if (.not.apply_upper_bc) return
167
168 call cnst_get_ind('F', f_ndx, abort=.false.)
169 call cnst_get_ind('HF', hf_ndx, abort=.false.)
170
171 steps_per_day = 86400/get_step_size()
172 zonal_avg = .false.
173
174!-----------------------------------------------------------------------
175! ... initialize the tgcm upper boundary module
176!-----------------------------------------------------------------------
177 call tgcm_ubc_inti( tgcm_ubc_file, tgcm_ubc_data_type, tgcm_ubc_cycle_yr, tgcm_ubc_fixed_ymd, tgcm_ubc_fixed_tod)
178 if (masterproc) write(iulog,*) 'ubc_init: after tgcm_ubc_inti'
179
180!-----------------------------------------------------------------------
181! ... initialize the snoe module
182!-----------------------------------------------------------------------
183 call snoe_inti(snoe_ubc_file)
184 if (masterproc) write(iulog,*) 'ubc_init: after snoe_inti'
185
186!-----------------------------------------------------------------------
187! ... initialize the msis module
188!-----------------------------------------------------------------------
189 call msis_ubc_inti( zonal_avg, steps_per_day )
190 if (masterproc) write(iulog,*) 'ubc_init: after msis_ubc_inti'
191
192 end subroutine ubc_init

225 subroutine ubc_get_vals (lchnk, ncol, ntop_molec, pint, zi, msis_temp, ubc_mmr, ubc_flux)
226!-----------------------------------------------------------------------
227! interface routine for vertical diffusion and pbl scheme
228!-----------------------------------------------------------------------
229 use mo_msis_ubc, only: get_msis_ubc
230 use mo_snoe, only: set_no_ubc, ndx_no
231 use mo_tgcm_ubc, only: set_tgcm_ubc
232 use cam_abortutils, only: endrun
233 use physconst, only: rairv, mbarv
234 use phys_control, only: waccmx_is
235 use physconst, only: avogad
236 use constituents, only: cnst_get_ind, cnst_mw ! Needed for ubc_flux
237
238!------------------------------Arguments--------------------------------
239 integer, intent(in) :: lchnk ! chunk identifier
240 integer, intent(in) :: ncol ! number of atmospheric columns
241 integer, intent(in) :: ntop_molec ! top of molecular diffusion region (=1)
242 real(r8), intent(in) :: pint(pcols,pverp) ! interface pressures
243 real(r8), intent(in) :: zi(pcols,pverp) ! interface geoptl height above sfc
244
245 real(r8), intent(out) :: ubc_mmr(pcols,pcnst) ! upper bndy mixing ratios (kg/kg)
246 real(r8), intent(out) :: msis_temp(pcols) ! upper bndy temperature (K)
247 real(r8), intent(out) :: ubc_flux(pcnst) ! upper bndy flux (kg/s/m^2)
248
249!---------------------------Local storage-------------------------------
250 integer :: ierr ! error flag for allocates
251 integer :: indx_H ! cnst index for H
252
253 real(r8), parameter :: m2km = 1.e-3_r8 ! meter to km
254 real(r8) :: rho_top(pcols) ! density at top interface
255 real(r8) :: z_top(pcols) ! height of top interface (km)
256
257!-----------------------------------------------------------------------
258
259 if (.not.apply_upper_bc) return
260
261 ubc_mmr(:,:) = 0._r8
262 ubc_flux(:) = 0._r8
263
264 call get_msis_ubc( lchnk, ncol, msis_temp, ubc_mmr )
265 if( t_pert_ubc /= 0._r8 ) then
266 msis_temp(:ncol) = msis_temp(:ncol) + t_pert_ubc
267 if( any( msis_temp(:ncol) < 0._r8 ) ) then
268 write(iulog,*) 'ubc_get_vals: msis temp < 0 after applying offset = ',t_pert_ubc
269 call endrun
270 end if
271 end if
272
273
274 !--------------------------------------------------------------------------------------------
275 ! For WACCM-X, calculate upper boundary H flux
276 !--------------------------------------------------------------------------------------------
277 if ( waccmx_is('ionosphere') .or. waccmx_is('neutral') ) then
278
279 call cnst_get_ind('H', indx_H)
280
281 ubc_flux(indx_H) = cnst_mw(indx_H)/ avogad * 0.9e12_r8 ! upward mass flux of H, kg/s/m^2
282
283 ubc_mmr(:ncol,ndx_no) = 0.0_r8
284
285 else ! for waccm
286
287 rho_top(:ncol) = pint(:ncol,ntop_molec) / (rairv(:ncol,ntop_molec,lchnk)*msis_temp(:ncol))
288 z_top(:ncol) = m2km * zi(:ncol,ntop_molec)
289
290 call set_no_ubc ( lchnk, ncol, z_top, ubc_mmr, rho_top )
291 if( ndx_no > 0 .and. no_xfac_ubc /= 1._r8 ) then
292 ubc_mmr(:ncol,ndx_no) = no_xfac_ubc * ubc_mmr(:ncol,ndx_no)
293 end if
294
295 endif
296
297 call set_tgcm_ubc( lchnk, ncol, ubc_mmr, mbarv(:,ntop_molec,lchnk))
298
299 if (f_ndx .GT. 0) then
300 ubc_mmr(:ncol, f_ndx) = 1.0e-15_r8
301 endif
302 if (hf_ndx .GT. 0) then
303 ubc_mmr(:ncol, hf_ndx) = 1.0e-15_r8
304 endif
305
306 end subroutine ubc_get_vals
