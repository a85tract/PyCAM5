732 subroutine makoh_cubic( cx, p2, p1, p0, im )
733!
734! solves x**3 + p2 x**2 + p1 x + p0 = 0
735! where p0, p1, p2 are real
736!

945 end do
946 den1 = dnc0(i-1) + dnc1(i-1) * t
947 den2 = dnc0(i) + dnc1(i) * t
948 ! Calculate derivative needed later for Kelvin factor for H2O
949 drho_dwt = (den2-den1) / (dnwtp(i)-dnwtp(i-1))
950 sulfate_density = den1 + drho_dwt * (wtpct_flat-dnwtp(i-1))
951
952 r=dmean*100._r8/2._r8 ! calcuate mode radius (cm) from diameter (m)
953
954 ! Adjust for Kelvin effect for water
955 rkelvinH2O_b = 1 + wtpct_flat * drho_dwt / &
956 sulfate_density - 3._r8 * wtpct_flat * dsigma_dwt / (2._r8*surf_tens)
957
958 rkelvinH2O_a = 2._r8 * wtmol_h2so4 * surf_tens / &
959 (sulfate_density * RGAS * t * r)
960
961 rkelvinH2O = exp (rkelvinH2O_a*rkelvinH2O_b)
962
963 qh2o_kelvin = qh2o/rkelvinH2O
964 call calc_h2so4_wtpct(t, pres, qh2o_kelvin, wtpct)
965
966
967 wtpct=max(wtpct,wtpct_flat)
968
969 ! Parameterized fit to Giauque's (1959) enthalpies v. wt %:
970 en = 4.184_r8 * (23624.8_r8 - 1.14208e8_r8 / ((wtpct - 105.318_r8)**2 + 4798.69_r8))
971 en = max(en, 0.0_r8)
972
973 !! Calculate surface tension (erg/cm2) of sulfate of
974 !! different compositions as a linear function of temperature.
975 i = 2 ! minimum wt% is 29.517
976 do while (wtpct.gt.stwtp(i))
977 i=i+1
978 end do
979 sig1=stc0(i-1)+stc1(i-1)*t
980 sig2=stc0(i)+stc1(i)*t
981 frac=(stwtp(i)-wtpct)/(stwtp(i)-stwtp(i-1))
982 surf_tens_mode=sig1*frac+sig2*(1.0_r8-frac)
983
984 !! Calculate density (g/cm3) of sulfate of
985 !! different compositions as a linear function of temperature.
986 i = 6 ! minimum wt% is 29.517
987 do while (wtpct .gt. dnwtp(i))
988 i=i+1
989 end do
990 den1=dnc0(i-1)+dnc1(i-1)*t
991 den2=dnc0(i)+dnc1(i)*t
992 frac=(dnwtp(i)-wtpct)/(dnwtp(i)-dnwtp(i-1))
993 sulden=den1*frac+den2*(1.0_r8-frac)
994
995 ! Ayers' (1980) fit to sulfuric acid equilibrium vapor pressure:
996 ! (Remember this is the log)
997 ! SULFEQ=-10156/Temp+16.259-En/(8.3143*Temp)
998 !
999 ! Kulmala correction (J. CHEM. PHYS. V.93, No.1, 1 July 1990)
1000 fk1 = -1._r8 / t
1001 fk4_1 = log(t0_kulm / t)
1002 fk4_2 = t0_kulm / t
1003 fk4 = 1.0_r8 + fk4_1 - fk4_2
1004 factor_kulm = fk1 + fk2 + fk3 * fk4
1005
1006 ! This is for pure H2SO4
1007 sulfequil = fk0 + 10156._r8 * factor_kulm
1008
1009 ! Adjust for WTPCT composition:
1010 sulfequil = sulfequil - en / (8.3143_r8 * t)
1011
1012 ! Take the exponential:
1013 sulfequil = exp(sulfequil)
1014
1015 ! Convert atmospheres ==> Pa
1016 sulfequil = sulfequil * 1.01325e5_r8
1017
1018 ! Convert Pa ==> mol/mol
1019 sulfequil = sulfequil / pres
1020
1021 ! Calculate Kelvin curvature factor for H2SO4 interactively with temperature:
1022 ! (g/mol)*(erg/cm2)/(K * g/cm3 * erg/mol/K) = cm
1023 akelvin = 2._r8 * wtmol_h2so4 * surf_tens_mode / (t * sulden * RGAS)
1024
1025 expon = akelvin / r ! divide by mode radius (cm)
1026 expon = max(-100._r8, expon)
1027 expon = min(100._r8, expon)
1028 akas = exp( expon )
1029 qh2so4_equilib = sulfequil * akas ! reduce H2SO4 equilibrium mixing ratio by Kelvin curvature factor
1030
1031 return
1032 end subroutine calc_h2so4_equilib_mixrat

1036 subroutine calc_h2so4_wtpct( temp, pres, qh2o, wtpct )
1037
1038 !! This function calculates the weight % H2SO4 composition of
1039 !! sulfate aerosol, using Tabazadeh et. al. (GRL, 1931, 1997).
1040 !! Rated for T=185-260K, activity=0.01-1.0
1041 !!
1042 !! Argument list input:
1043 !! temp = temperature (K)
1044 !! pres = atmospheric pressure (Pa)
1045 !! qh2o = water specific humidity (kg/kg)
1046 !!
1047 !! Output:
1048 !! wtpct = weight % H2SO4 in H2O/H2SO4 particle (0-100)
1049 !!
1050 !! @author Mike Mills
1051 !! @ version October 2013
1052
1053 use wv_saturation, only: qsat_water
1054
1055 implicit none
1056
1057 real(r8), intent(in) :: temp ! temperature (K)
1058 real(r8), intent(in) :: pres ! pressure (Pa)
1059 real(r8), intent(in) :: qh2o ! water vapor specific humidity (kg/kg)
1060 real(r8), intent(out) :: wtpct ! sulfate weight % H2SO4 composition
1061
1062 ! Local declarations
1063 real(r8) :: atab1,btab1,ctab1,dtab1,atab2,btab2,ctab2,dtab2
1064 real(r8) :: contl, conth, contt, conwtp
1065 real(r8) :: activ
1066 real(r8) :: es ! saturation vapor pressure over water (Pa) (dummy)
1067 real(r8) :: qs ! saturation specific humidity over water (kg/kg)
1068
1069 ! calculate saturation specific humidity over pure water, qs (kg/kg)
1070 call qsat_water(temp, pres, es, qs)
1071
1072 ! Activity = water specific humidity (kg/kg) / equilibrium water (kg/kg)
1073 activ = qh2o/qs
1074
1075 if (activ.lt.0.05_r8) then
1076 activ = max(activ,1.e-6_r8) ! restrict minimum activity
1077 atab1 = 12.37208932_r8
1078 btab1 = -0.16125516114_r8
1079 ctab1 = -30.490657554_r8
1080 dtab1 = -2.1133114241_r8
1081 atab2 = 13.455394705_r8
1082 btab2 = -0.1921312255_r8
1083 ctab2 = -34.285174607_r8
1084 dtab2 = -1.7620073078_r8
1085 elseif (activ.ge.0.05_r8.and.activ.le.0.85_r8) then
1086 atab1 = 11.820654354_r8
1087 btab1 = -0.20786404244_r8
1088 ctab1 = -4.807306373_r8
1089 dtab1 = -5.1727540348_r8
1090 atab2 = 12.891938068_r8
1091 btab2 = -0.23233847708_r8
1092 ctab2 = -6.4261237757_r8
1093 dtab2 = -4.9005471319_r8
1094 elseif (activ.gt.0.85_r8) then
1095 activ = min(activ,1._r8) ! restrict maximum activity
1096 atab1 = -180.06541028_r8
1097 btab1 = -0.38601102592_r8
1098 ctab1 = -93.317846778_r8
1099 dtab1 = 273.88132245_r8
1100 atab2 = -176.95814097_r8
1101 btab2 = -0.36257048154_r8
1102 ctab2 = -90.469744201_r8
1103 dtab2 = 267.45509988_r8
1104 else
1105 write(iulog,*) 'calc_h2so4_wtpct: invalid activity: activ,qh2o,qs,temp,pres=',activ,qh2o,qs,temp,pres
1106 call endrun( 'calc_h2so4_wtpct error' )
1107 return
1108 endif
1109
1110 contl = atab1*(activ**btab1)+ctab1*activ+dtab1
1111 conth = atab2*(activ**btab2)+ctab2*activ+dtab2
1112
1113 contt = contl + (conth-contl) * ((temp -190._r8)/70._r8)
1114 conwtp = (contt*98._r8) + 1000._r8
1115
1116 wtpct = (100._r8*contt*98._r8)/conwtp
1117 wtpct = min(max(wtpct,25._r8),100._r8) ! restrict between 1 and 100 %
1118
1119 return
1120 end subroutine calc_h2so4_wtpct
