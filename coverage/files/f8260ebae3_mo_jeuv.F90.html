326 subroutine jeuv( nlev, zen, occ, o2cc, n2cc, &
327 zkm, euv_prates)
328!==============================================================================
329! Purpose:
330! Calculate euv photolysis/ionization rates (in s-1)
331!==============================================================================
332! Arguments:
333! nlev: number of model vertical levels
334! zen: solar zenith angle in degree
335! occ: atmoic oxygen number density (#/cm3)
336! o2cc: molecular oxygen number density (#/cm3)
337! n2cc: molecular nitrogen number density (#/cm3)
338! zkm: altitude of model levels in KM
339! euv_prates: array for EUV photolysis/ionization rates
340!==============================================================================
341! Approach:
342! call sphers
343! input: zenith angle
344! output: dsdh and nid used in slant column routine
345! call slant_col
346! input: dsdh and nid, etc
347! output: slant column density
348! calculate photon production rates and photoelectron production rates
349!==============================================================================
350! Photolysis/ionization considered in EUV calculation
351!
352! O + hv --> O+ (4S) + e* ! J1
353! O + hv --> O+ (2D) + e* ! J2
354! O + hv --> O+ (2P) + e* ! J3
355! N (4S) + hv --> N+ + e* ! J4
356! O2 + hv --> O2+ + e* ! J5
357! N2 + hv --> N2+ + e* ! J6
358! O2 + hv --> O + O+(4S) + e* ! J7
359! O2 + hv --> O + O+(2D) + e* ! J8
360! O2 + hv --> O + O+(2P) + e* ! J9
361! N2 + hv --> N (4S) + N+ + e* ! J10
362! N2 + hv --> N (2D) + N+ + e* ! J11
363! O2 + hv --> O (3P) + O (3P) ! J12
364! N2 + hv --> N (4S) + N (2D) ! J13
365!
366! O + e* --> O+ (4S) + e* + e ! J14
367! O + e* --> O+ (2D) + e*+ e ! J15
368! O + e* --> O+ (2P) + e*+ e ! J16
369! O2 + e* --> O2+ + e*+ e ! J17
370! N2 + e*--> N2+ + e*+ e ! J18
371! O2 + e* --> O + O+(4S) + e*+ e ! J19
372! O2 + e*--> O + O+(2D) + e*+ e ! J20
373! O2 + e* --> O + O+(2P) + e*+ e ! J21
374! N2 + e* --> N (4S) + N+ + e*+ e ! J22
375! N2 + e* --> N (2D) + N+ + e*+ e ! J23
376! O2 + e* --> O (3P) + O (3P) + e* ! J24
377! N2 + e* --> N (4S) + N (2D) + e* ! J25
378!==============================================================================
379
380 use euvac, only : euvac_etf
381 use mo_jshort, only : sphers, slant_col
382 use cam_history, only : outfld
383
384 implicit none
385
386!------------------------------------------------------------------------------
387! dummy arguments
388!------------------------------------------------------------------------------
389 integer, intent(in) :: nlev ! model vertical levels
390 real(r8), intent(in) :: zen ! Zenith angle in degree
391 real(r8), intent(in) :: occ(nlev) ! atmic oxygen number density (#/cm3)
392 real(r8), intent(in) :: o2cc(nlev) ! Molecular oxygen number density (#/cm3)
393 real(r8), intent(in) :: n2cc(nlev) ! molecular nitrogen number density(#/cm3)
394 real(r8), intent(in) :: zkm(nlev) ! Altitude, km,from top to bottom
395 real(r8), intent(out) :: euv_prates(:,:) ! EUV photolysis/ionization rates (s-1)
396
397!------------------------------------------------------------------------------
398! local variables
399!------------------------------------------------------------------------------
400 real(r8), parameter :: km2cm = 1.e5_r8
401 integer :: l, k, m, n ! loop indecies
402 real(r8) :: tau(lmax) ! wavelength dependant optical depth
403 real(r8) :: delz(nlev) ! layer thickness (cm)
404 real(r8) :: scol(nlev,nmaj) ! major species' (O,O2,N2) Slant Column Density
405 real(r8) :: nflux(nlev,lmax)
406 real(r8) :: wrk(nmaj) ! temporary array for photoabsorption rate
407 real(r8) :: absorp(nlev,lmax) ! temporary array for photoabsorption rate
408 real(r8) :: ioniz(nlev,lmax) ! temporary array for photoionization rate
409 real(r8) :: dsdh(0:nlev,nlev) ! Slant path of direct beam through each layer
410 ! crossed when travelling from the top of the atmosphere
411 ! to layer i; dsdh(i,j), i = 0..NZ-1, j = 1..NZ-1
412 integer :: nid(0:nlev) ! Number of layers crossed by the direct
413 ! beam when travelling from the top of the
414 ! atmosphere to layer i; NID(i), i = 0..NZ-1
415 real(r8) :: p_photon(nlev,nstat,nspecies) ! photoionization/dissociation rates(s-1) (O,O2,N2,N)
416 real(r8) :: p_electron(nlev,nstat,nmaj) ! photoelectron ionization/dissociation rates(s-1) (O,O2,N2)
417
418 real(r8) :: prates(nlev,neuv)
419!------------------------------------------------------------------------------
420! zero arrays
421!------------------------------------------------------------------------------
422 p_photon(:,:,:) = 0._r8
423 p_electron(:,:,:) = 0._r8
424
425 call sphers( nlev, zkm, zen, dsdh, nid )
426 delz(1:nlev-1) = km2cm*(zkm(1:nlev-1) - zkm(2:nlev))
427 call slant_col( nlev, delz, dsdh, nid, occ, scol )
428 call slant_col( nlev, delz, dsdh, nid, o2cc, scol(1,2) )
429 call slant_col( nlev, delz, dsdh, nid, n2cc, scol(1,3) )
430
431!------------------------------------------------------------------------------
432! The calculation is in the order from model bottom to model top
433! because scol array is in this order.
434!------------------------------------------------------------------------------
435 do k = 1,nlev
436 wrk(:) = scol(k,:)
437 tau(:) = matmul( sigabs(:,:nmaj),wrk )
438 where( tau(:) < 20._r8 )
439 nflux(k,:) = euvac_etf(:) * exp( -tau(:) )
440 elsewhere
441 nflux(k,:) = 0._r8
442 endwhere
443 end do
444
445!------------------------------------------------------------------------------
446! remember occ,o2cc and n2cc is from top to bottom
447!------------------------------------------------------------------------------
448 do m = 1,nspecies
449 do l = 1,lmax
450 absorp(:,l) = sigabs(l,m) * nflux(:,l)
451 end do
452 if( m <= nmaj ) then
453 do l = 1,lmax
454 ioniz(:,l) = absorp(:,l) * branch_p(l,1,m)
455 end do
456 do n = 1,nstat
457 p_photon(:,n,m) = matmul( absorp,branch_p(:,n,m) )
458 p_electron(:,n,m) = matmul( ioniz,branch_e(:,n,m) )
459 end do
460 else
461 p_photon(:,1,m) = matmul( nflux,sigabs(:,m) )
462 end if
463 end do
464
465!------------------------------------------------------------------------------
466! set photolysis/ionization rate for each reaction
467!------------------------------------------------------------------------------
468 prates(:,1) = p_photon(:,2,1)
469 prates(:,2) = p_photon(:,3,1)
470 prates(:,3) = p_photon(:,4,1)
471 prates(:,4) = p_photon(:,1,4)
472 prates(:,5) = p_photon(:,2,2) + p_photon(:,3,2)
473 prates(:,6) = p_photon(:,2,3) + p_photon(:,3,3)
474 prates(:,7) = .54_r8*p_photon(:,4,2)
475 prates(:,8) = .24_r8*p_photon(:,4,2)
476 prates(:,9) = .22_r8*p_photon(:,4,2)
477 prates(:,10) = .2_r8*p_photon(:,4,3)
478 prates(:,11) = .8_r8*p_photon(:,4,3)
479 prates(:,12) = p_photon(:,5,2)
480 prates(:,13) = p_photon(:,5,3)
481 prates(:,14) = p_electron(:,2,1)
482 prates(:,15) = p_electron(:,3,1)
483 prates(:,16) = p_electron(:,4,1)
484 prates(:,17) = p_electron(:,2,2) + p_electron(:,3,2)
485 prates(:,18) = p_electron(:,2,3) + p_electron(:,3,3)
486 prates(:,19) = .54_r8*p_electron(:,4,2)
487 prates(:,20) = .24_r8*p_electron(:,4,2)
488 prates(:,21) = .22_r8*p_electron(:,4,2)
489 prates(:,22) = .2_r8*p_electron(:,4,3)
490 prates(:,23) = .8_r8*p_electron(:,4,3)
491 prates(:,24) = p_electron(:,5,2)
492 prates(:,25) = p_electron(:,5,3)
493 prates(:,26) = p_photon(:,1,5)
494
495 do m = 1,neuv
496 euv_prates(:,m) = prates(nlev:1:-1,m)
497 enddo
498
499 end subroutine jeuv

550!------------------------------------------------------------------------------
551 integer, intent(in) :: nlev ! model vertical levels
552 integer, intent(in) :: kbot ! heating vertical levels
553 real(r8), intent(in) :: zen ! zenith angle (degrees)
554 real(r8), intent(in) :: occ(nlev) ! atomic oxygen number density (molecules/cm^3)
555 real(r8), intent(in) :: o2cc(nlev) ! molecular oxygen concentration (molecules/cm^3)
556 real(r8), intent(in) :: n2cc(nlev) ! molecular nitrogen concentration (molecules/cm^3)
557 real(r8), intent(in) :: o_vmr(nlev) ! atomic oxygen concentration (mol/mol)
558 real(r8), intent(in) :: o2_vmr(nlev) ! molecular oxygen concentration (mol/mol)
559 real(r8), intent(in) :: n2_vmr(nlev) ! molecular nitrogen concentration (mol/mol)
560 real(r8), intent(in) :: zkm(nlev) ! midpoint geopotential (km)
561 real(r8), intent(in) :: cparg(nlev) ! specific heat capacity
562 real(r8), intent(in) :: mw(nlev) ! atm mean mass
563 real(r8), intent(out) :: euv_hrates(:) ! euv heating rates
564
565!------------------------------------------------------------------------------
566! local variables
567!------------------------------------------------------------------------------
568 real(r8), parameter :: km2cm = 1.e5_r8
569 integer :: l, k, k1, m, n ! indicies
570 real(r8) :: tau(lmax) ! wavelength dependant optical depth
571 real(r8) :: delz(nlev) ! layer thickness (cm)
572 real(r8) :: hfactor(kbot)
573 real(r8) :: scol(nlev,nmaj) ! major species' (O,O2,N2) Slant Column Density
574 real(r8) :: nflux(kbot,lmax)
575 real(r8) :: prates(kbot,13) ! working photorates array
576 real(r8) :: wrk(nmaj) ! temporary array for photoabsorption rate
577 real(r8) :: absorp(kbot,lmax) ! temporary array for photoabsorption rate
578 real(r8) :: ioniz(kbot,lmax) ! temporary array for photoionization rate
579 real(r8) :: dsdh(0:nlev,nlev) ! Slant path of direct beam through each layer
580 ! crossed when travelling from the top of the atmosphere
581 ! to layer i; dsdh(i,j), i = 0..NZ-1, j = 1..NZ-1
582 integer :: nid(0:nlev) ! Number of layers crossed by the direct
583 ! beam when travelling from the top of the
584 ! atmosphere to layer i; NID(i), i = 0..NZ-1
585 real(r8) :: p_photon(kbot,nstat,nspecies) ! photoionization/dissociation rates(s-1) (O,O2,N2,N)
586
587!------------------------------------------------------------------------------
588! zero arrays
589!------------------------------------------------------------------------------
590 p_photon(:,:,:) = 0._r8
591
592 call sphers( nlev, zkm, zen, dsdh, nid )
593 delz(1:nlev-1) = km2cm*(zkm(1:nlev-1) - zkm(2:nlev))
594 call slant_col( nlev, delz, dsdh, nid, occ, scol )
595 call slant_col( nlev, delz, dsdh, nid, o2cc, scol(1,2) )
596 call slant_col( nlev, delz, dsdh, nid, n2cc, scol(1,3) )
597
598!------------------------------------------------------------------------------
599! The calculation is in the order from model bottom to model top
600! because scol array is in this order.
601!------------------------------------------------------------------------------
602 do k = 1,kbot
603 k1 = nlev - kbot + k
604 wrk(:) = scol(k1,:)
605 tau(:) = matmul( sigabs(:,:nmaj),wrk )
606 where( tau(:) < 20._r8 )
607 nflux(k,:) = energy(:) * euvac_etf(:) * exp( -tau(:) )
608 elsewhere
609 nflux(k,:) = 0._r8
610 endwhere
611 end do
612
613!------------------------------------------------------------------------------
614! remember occ,o2cc and n2cc is from top to bottom
615!------------------------------------------------------------------------------
616 do m = 1,nspecies
617 do l = 1,lmax
618 absorp(:,l) = sigabs(l,m) * nflux(:,l)
619 end do
620 if( m <= nmaj ) then
621 do n = 1,nstat
622 p_photon(:,n,m) = matmul( absorp,branch_p(:,n,m) )
623 end do
624 else
625 p_photon(:,1,m) = matmul( nflux,sigabs(:,m) )
626 end if
627 end do
628
629!------------------------------------------------------------------------------
630! set photolysis rate
631!------------------------------------------------------------------------------
632 prates(:,1) = p_photon(:,2,1)
633 prates(:,2) = p_photon(:,3,1)
634 prates(:,3) = p_photon(:,4,1)
635 prates(:,5) = p_photon(:,2,2) + p_photon(:,3,2)
636 prates(:,6) = p_photon(:,2,3) + p_photon(:,3,3)
637 prates(:,7) = .54_r8*p_photon(:,4,2)
638 prates(:,8) = .24_r8*p_photon(:,4,2)
639 prates(:,9) = .22_r8*p_photon(:,4,2)
640 prates(:,10) = .2_r8*p_photon(:,4,3)
641 prates(:,11) = .8_r8*p_photon(:,4,3)
642 prates(:,12) = p_photon(:,5,2)
643 prates(:,13) = p_photon(:,5,3)
644 hfactor(:) = avogad/(cparg(:kbot)*mw(:kbot))
645 euv_hrates(kbot+1:nlev) = 0._r8
646 euv_hrates(:kbot) = ((prates(kbot:1:-1,1) + prates(kbot:1:-1,2) + prates(kbot:1:-1,3))*o_vmr(:kbot) &
647 + (prates(kbot:1:-1,5) + prates(kbot:1:-1,7) + prates(kbot:1:-1,8) &
648 + prates(kbot:1:-1,9) + prates(kbot:1:-1,12))*o2_vmr(:kbot) &
649 + (prates(kbot:1:-1,6) + prates(kbot:1:-1,10) &
650 + prates(kbot:1:-1,11) + prates(kbot:1:-1,13))*n2_vmr(:kbot))*hfactor(:)
651
652 end subroutine heuv
653
654 end module mo_jeuv
