655! contact freezing (-40<T<-3 C) (Young, 1974) with hooks into simulated dust
656!===================================================================
657! dust size and number in multiple bins are read in from companion routine
658
659pure subroutine contact_freezing (microp_uniform, t, p, rndst, nacon, &
660 pgam, lamc, cdist1, qcic, relvar, mnucct, nnucct)
661
662 logical, intent(in) :: microp_uniform
663
664 real(r8), intent(in) :: t(:) ! Temperature
665 real(r8), intent(in) :: p(:) ! Pressure
666 real(r8), intent(in) :: rndst(:,:) ! Radius (for multiple dust bins)
667 real(r8), intent(in) :: nacon(:,:) ! Number (for multiple dust bins)
668
669 ! Size distribution parameters for cloud droplets
670 real(r8), intent(in) :: pgam(:)
671 real(r8), intent(in) :: lamc(:)
672 real(r8), intent(in) :: cdist1(:)
673
674 ! MMR of in-cloud liquid water
675 real(r8), intent(in) :: qcic(:)
676
677 ! Relative cloud water variance
678 real(r8), intent(in) :: relvar(:)
679
680 ! Output tendencies
681 real(r8), intent(out) :: mnucct(:) ! MMR
682 real(r8), intent(out) :: nnucct(:) ! Number
683
684 real(r8) :: tcnt ! scaled relative temperature
685 real(r8) :: viscosity ! temperature-specific viscosity (kg/m/s)
686 real(r8) :: mfp ! temperature-specific mean free path (m)
687
688 ! Dimension these according to number of dust bins, inferred from rndst size
689 real(r8) :: nslip(size(rndst,2)) ! slip correction factors
690 real(r8) :: ndfaer(size(rndst,2)) ! aerosol diffusivities (m^2/sec)
691
692 ! Coefficients not used for subcolumns
693 real(r8) :: dum, dum1
694
695 integer :: i
696
697 do i = 1,size(t)
698
699 if (qcic(i) >= qsmall .and. t(i) < 269.15_r8) then
700
701 if (.not. microp_uniform) then
702 dum = var_coef(relvar(i), 4._r8/3._r8)
703 dum1 = var_coef(relvar(i), 1._r8/3._r8)
704 else
705 dum = 1._r8
706 dum1 = 1._r8
707 endif
708
709 tcnt=(270.16_r8-t(i))**1.3_r8
710 viscosity = 1.8e-5_r8*(t(i)/298.0_r8)**0.85_r8 ! Viscosity (kg/m/s)
711 mfp = 2.0_r8*viscosity/ & ! Mean free path (m)
712 (p(i)*sqrt( 8.0_r8*28.96e-3_r8/(pi*8.314409_r8*t(i)) ))
713
714 ! Note that these two are vectors.
715 nslip = 1.0_r8+(mfp/rndst(i,:))*(1.257_r8+(0.4_r8*exp(-(1.1_r8*rndst(i,:)/mfp))))! Slip correction factor
716
717 ndfaer = 1.381e-23_r8*t(i)*nslip/(6._r8*pi*viscosity*rndst(i,:)) ! aerosol diffusivity (m2/s)
718
719 mnucct(i) = dum * &
720 dot_product(ndfaer,nacon(i,:)*tcnt)*pi*pi/3._r8*rhow* &
721 cdist1(i)*gamma(pgam(i)+5._r8)/lamc(i)**4
722
723 nnucct(i) = dum1 * &
724 dot_product(ndfaer,nacon(i,:)*tcnt)*2._r8*pi* &
725 cdist1(i)*gamma(pgam(i)+2._r8)/lamc(i)
726
727 else
728
729 mnucct(i)=0._r8
730 nnucct(i)=0._r8
731
732 end if ! qcic > qsmall and t < 4 deg C
733 end do
734
735end subroutine contact_freezing

767! accretion of cloud droplets onto snow/graupel
768!===================================================================
769! here use continuous collection equation with
770! simple gravitational collection kernel
771! ignore collisions between droplets/cloud ice
772! since minimum size ice particle for accretion is 50 - 150 micron
773
774elemental subroutine accrete_cloud_water_snow(t, rho, asn, uns, mu, qcic, ncic, qsic, &
775 pgam, lamc, lams, n0s, psacws, npsacws)
776
777 real(r8), intent(in) :: t ! Temperature
778 real(r8), intent(in) :: rho ! Density
779 real(r8), intent(in) :: asn ! Fallspeed parameter (snow)
780 real(r8), intent(in) :: uns ! Current fallspeed (snow)
781 real(r8), intent(in) :: mu ! Viscosity
782
783 ! In-cloud liquid water
784 real(r8), intent(in) :: qcic ! MMR
785 real(r8), intent(in) :: ncic ! Number
786
787 ! In-cloud snow
788 real(r8), intent(in) :: qsic ! MMR
789
790 ! Cloud droplet size parameters
791 real(r8), intent(in) :: pgam
792 real(r8), intent(in) :: lamc
793
794 ! Snow size parameters
795 real(r8), intent(in) :: lams
796 real(r8), intent(in) :: n0s
797
798 ! Output tendencies
799 real(r8), intent(out) :: psacws ! Mass mixing ratio
800 real(r8), intent(out) :: npsacws ! Number concentration
801
802 real(r8) :: dc0 ! Provisional mean droplet size
803 real(r8) :: dum
804 real(r8) :: eci ! collection efficiency for riming of snow by droplets
805
806 ! ignore collision of snow with droplets above freezing
807
808 if (qsic >= qsmall .and. t <= tmelt .and. qcic >= qsmall) then
809
810 ! put in size dependent collection efficiency
811 ! mean diameter of snow is area-weighted, since
812 ! accretion is function of crystal geometric area
813 ! collection efficiency is approximation based on stoke's law (Thompson et al. 2004)
814
815 dc0 = (pgam+1._r8)/lamc
816 dum = dc0*dc0*uns*rhow/(9._r8*mu*(1._r8/lams))
817 eci = dum*dum/((dum+0.4_r8)*(dum+0.4_r8))
818
819 eci = max(eci,0._r8)
820 eci = min(eci,1._r8)
821
822 ! no impact of sub-grid distribution of qc since psacws
823 ! is linear in qc
824
825 psacws = pi/4._r8*asn*qcic*rho*n0s*eci*gamma_bs_plus3 / lams**(bs+3._r8)
826 npsacws = pi/4._r8*asn*ncic*rho*n0s*eci*gamma_bs_plus3 / lams**(bs+3._r8)
827 else
828 psacws = 0._r8
829 npsacws = 0._r8
830 end if

832end subroutine accrete_cloud_water_snow

834! add secondary ice production due to accretion of droplets by snow
835!===================================================================
836! (Hallet-Mossop process) (from Cotton et al., 1986)
837
838elemental subroutine secondary_ice_production(t, psacws, msacwi, nsacwi)
839 real(r8), intent(in) :: t ! Temperature
840
841 ! Accretion of cloud water to snow tendencies
842 real(r8), intent(inout) :: psacws ! MMR
843
844 ! Output (ice) tendencies
845 real(r8), intent(out) :: msacwi ! MMR
846 real(r8), intent(out) :: nsacwi ! Number
847
848 if((t < 270.16_r8) .and. (t >= 268.16_r8)) then
849 nsacwi = 3.5e8_r8*(270.16_r8-t)/2.0_r8*psacws

851 else if((t < 268.16_r8) .and. (t >= 265.16_r8)) then

855 nsacwi = 0.0_r8
856 msacwi = 0.0_r8
857 endif
858
859 psacws = max(0.0_r8,psacws - nsacwi*mi0)

1067! calculate evaporation/sublimation of rain and snow
1068!===================================================================
1069! note: evaporation/sublimation occurs only in cloud-free portion of grid cell
1070! in-cloud condensation/deposition of rain and snow is neglected
1071! except for transfer of cloud water to snow through bergeron process
1072
1073elemental subroutine evaporate_sublimate_precip(t, rho, dv, mu, sc, q, qvl, qvi, &
1074 lcldm, cldmax, arn, asn, qcic, qiic, qric, qsic, lamr, n0r, lams, n0s, &
1075 pre, prds)
1076
1077 real(r8), intent(in) :: t ! temperature
1078 real(r8), intent(in) :: rho ! air density
1079 real(r8), intent(in) :: dv ! water vapor diffusivity
1080 real(r8), intent(in) :: mu ! viscosity
1081 real(r8), intent(in) :: sc ! schmidt number
1082 real(r8), intent(in) :: q ! humidity
1083 real(r8), intent(in) :: qvl ! saturation humidity (water)
1084 real(r8), intent(in) :: qvi ! saturation humidity (ice)
1085 real(r8), intent(in) :: lcldm ! liquid cloud fraction
1086 real(r8), intent(in) :: cldmax ! precipitation fraction (maximum overlap)
1087
1088 ! fallspeed parameters
1089 real(r8), intent(in) :: arn ! rain
1090 real(r8), intent(in) :: asn ! snow
1091
1092 ! In-cloud MMRs
1093 real(r8), intent(in) :: qcic ! cloud liquid
1094 real(r8), intent(in) :: qiic ! cloud ice
1095 real(r8), intent(in) :: qric ! rain
1096 real(r8), intent(in) :: qsic ! snow
1097
1098 ! Size parameters
1099 ! rain
1100 real(r8), intent(in) :: lamr
1101 real(r8), intent(in) :: n0r
1102 ! snow
1103 real(r8), intent(in) :: lams
1104 real(r8), intent(in) :: n0s
1105
1106 ! Output tendencies
1107 real(r8), intent(out) :: pre
1108 real(r8), intent(out) :: prds
1109
1110 real(r8) :: qclr ! water vapor mixing ratio in clear air
1111 real(r8) :: ab ! correction to account for latent heat
1112 real(r8) :: eps ! 1/ sat relaxation timescale
1113
1114 real(r8) :: dum
1115
1116 ! set temporary cloud fraction to zero if cloud water + ice is very small
1117 ! this will ensure that evaporation/sublimation of precip occurs over
1118 ! entire grid cell, since min cloud fraction is specified otherwise
1119 if (qcic+qiic < 1.e-6_r8) then
1120 dum = 0._r8
1121 else
1122 dum = lcldm
1123 end if
1124
1125 ! only calculate if there is some precip fraction > cloud fraction
1126
1127 if (cldmax > dum) then
1128
1129 ! calculate q for out-of-cloud region
1130 qclr=(q-dum*qvl)/(1._r8-dum)
1131
1132 ! evaporation of rain
1133 if (qric >= qsmall) then
1134
1135 ab = calc_ab(t, qvl, xxlv)
1136 eps = 2._r8*pi*n0r*rho*Dv* &
1137 (f1r/(lamr*lamr)+ &
1138 f2r*(arn*rho/mu)**0.5_r8* &
1139 sc**(1._r8/3._r8)*gamma_half_br_plus5/ &
1140 (lamr**(5._r8/2._r8+br/2._r8)))
1141
1142 pre = eps*(qclr-qvl)/ab
1143
1144 ! only evaporate in out-of-cloud region
1145 ! and distribute across cldmax
1146 pre=min(pre*(cldmax-dum),0._r8)
1147 pre=pre/cldmax
1148 else
1149 pre = 0._r8
1150 end if
1151
1152 ! sublimation of snow
1153 if (qsic >= qsmall) then
1154 ab = calc_ab(t, qvi, xxls)
1155 eps = 2._r8*pi*n0s*rho*Dv* &
1156 (f1s/(lams*lams)+ &
1157 f2s*(asn*rho/mu)**0.5_r8* &
1158 sc**(1._r8/3._r8)*gamma_half_bs_plus5/ &
1159 (lams**(5._r8/2._r8+bs/2._r8)))
1160 prds = eps*(qclr-qvi)/ab
1161
1162 ! only sublimate in out-of-cloud region and distribute over cldmax
1163 prds=min(prds*(cldmax-dum),0._r8)
1164 prds=prds/cldmax
1165 else
1166 prds = 0._r8
1167 end if
1168
1169 else
1170 prds = 0._r8
1171 pre = 0._r8
1172 end if
1173
1174end subroutine evaporate_sublimate_precip
