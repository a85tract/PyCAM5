11815222452224 subroutine physics_dme_adjust(state, tend, qini, dt)
1182 !-----------------------------------------------------------------------
1183 !
1184 ! Purpose: Adjust the dry mass in each layer back to the value of physics input state
1185 !
1186 ! Method: Conserve the integrated mass, momentum and total energy in each layer
1187 ! by scaling the specific mass of consituents, specific momentum (velocity)
1188 ! and specific total energy by the relative change in layer mass. Solve for
1189 ! the new temperature by subtracting the new kinetic energy from total energy
1190 ! and inverting the hydrostatic equation
1191 !
1192 ! The mass in each layer is modified, changing the relationship of the layer
1193 ! interfaces and midpoints to the surface pressure. The result is no longer in
1194 ! the original hybrid coordinate.
1195 !
1196 ! This procedure cannot be applied to the "eul" or "sld" dycores because they
1197 ! require the hybrid coordinate.
1198 !
1199 ! Author: Byron Boville
1200
1201 ! !REVISION HISTORY:
1202 ! 03.03.28 Boville Created, partly from code by Lin in p_d_adjust
1203 !

126000 te (:ncol) = te (:ncol) / fdq(:ncol)

126200 state%v(:ncol,k) = state%v(:ncol,k ) / fdq(:ncol)

126400 tend%dudt(:ncol,k) = (state%u(:ncol,k) - utmp(:ncol)) / dt
126500 tend%dvdt(:ncol,k) = (state%v(:ncol,k) - vtmp(:ncol)) / dt
