3 !----------------------------------------------------------------------------------------------------- !
4 ! Module to compute vertical diffusion of momentum, moisture, trace constituents !
5 ! and static energy. Separate modules compute !
6 ! 1. stresses associated with turbulent flow over orography !
7 ! ( turbulent mountain stress ) !
8 ! 2. eddy diffusivities, including nonlocal tranport terms !
9 ! 3. molecular diffusivities !
10 ! Lastly, a implicit diffusion solver is called, and tendencies retrieved by !
11 ! differencing the diffused and initial states. !
12 ! !
13 ! Calling sequence: !
14 ! !
15 ! vertical_diffusion_init Initializes vertical diffustion constants and modules !
16 ! init_molec_diff Initializes molecular diffusivity module !
17 ! init_eddy_diff Initializes eddy diffusivity module (includes PBL) !
18 ! init_tms Initializes turbulent mountain stress module !
19 ! init_vdiff Initializes diffusion solver module !
20 ! vertical_diffusion_ts_init Time step initialization (only used for upper boundary condition) !
21 ! vertical_diffusion_tend Computes vertical diffusion tendencies !
22 ! compute_tms Computes turbulent mountain stresses !
23 ! compute_eddy_diff Computes eddy diffusivities and countergradient terms !
24 ! compute_vdiff Solves vertical diffusion equations, including molecular diffusivities !
25 ! !
26 !---------------------------Code history-------------------------------------------------------------- !
27 ! J. Rosinski : Jun. 1992 !
28 ! J. McCaa : Sep. 2004 !
29 ! S. Park : Aug. 2006, Dec. 2008. Jan. 2010 !
30 !----------------------------------------------------------------------------------------------------- !

1391 subroutine positive_moisture( cp, xlv, xls, ncol, mkx, dt, qvmin, qlmin, qimin, &
1392 dp, qv, ql, qi, t, s, qvten, qlten, qiten, sten )
1393 ! ------------------------------------------------------------------------------- !
1394 ! If any 'ql < qlmin, qi < qimin, qv < qvmin' are developed in any layer, !
1395 ! force them to be larger than minimum value by (1) condensating water vapor !
1396 ! into liquid or ice, and (2) by transporting water vapor from the very lower !
1397 ! layer. '2._r8' is multiplied to the minimum values for safety. !
1398 ! Update final state variables and tendencies associated with this correction. !
1399 ! If any condensation happens, update (s,t) too. !
1400 ! Note that (qv,ql,qi,t,s) are final state variables after applying corresponding !
1401 ! input tendencies. !
1402 ! Be careful the order of k : '1': near-surface layer, 'mkx' : top layer !
1403 ! ------------------------------------------------------------------------------- !
1404 implicit none
1405 integer, intent(in) :: ncol, mkx
1406 real(r8), intent(in) :: cp, xlv, xls
1407 real(r8), intent(in) :: dt, qvmin, qlmin, qimin
1408 real(r8), intent(in) :: dp(ncol,mkx)
1409 real(r8), intent(inout) :: qv(ncol,mkx), ql(ncol,mkx), qi(ncol,mkx), t(ncol,mkx), s(ncol,mkx)
1410 real(r8), intent(inout) :: qvten(ncol,mkx), qlten(ncol,mkx), qiten(ncol,mkx), sten(ncol,mkx)
1411 integer i, k
1412 real(r8) dql, dqi, dqv, sum, aa, dum
1413
1414 ! Modification : I should check whether this is exactly same as the one used in
1415 ! shallow convection and cloud macrophysics.
1416
1417 do i = 1, ncol
1418 do k = mkx, 1, -1 ! From the top to the 1st (lowest) layer from the surface
1419 dql = max(0._r8,1._r8*qlmin-ql(i,k))
1420 dqi = max(0._r8,1._r8*qimin-qi(i,k))
1421 qlten(i,k) = qlten(i,k) + dql/dt
1422 qiten(i,k) = qiten(i,k) + dqi/dt
1423 qvten(i,k) = qvten(i,k) - (dql+dqi)/dt
1424 sten(i,k) = sten(i,k) + xlv * (dql/dt) + xls * (dqi/dt)
1425 ql(i,k) = ql(i,k) + dql
1426 qi(i,k) = qi(i,k) + dqi
1427 qv(i,k) = qv(i,k) - dql - dqi
1428 s(i,k) = s(i,k) + xlv * dql + xls * dqi
1429 t(i,k) = t(i,k) + (xlv * dql + xls * dqi)/cp
1430 dqv = max(0._r8,1._r8*qvmin-qv(i,k))
1431 qvten(i,k) = qvten(i,k) + dqv/dt
1432 qv(i,k) = qv(i,k) + dqv
1433 if( k .ne. 1 ) then
1434 qv(i,k-1) = qv(i,k-1) - dqv*dp(i,k)/dp(i,k-1)
1435 qvten(i,k-1) = qvten(i,k-1) - dqv*dp(i,k)/dp(i,k-1)/dt
1436 endif
1437 qv(i,k) = max(qv(i,k),qvmin)
1438 ql(i,k) = max(ql(i,k),qlmin)
1439 qi(i,k) = max(qi(i,k),qimin)
1440 end do

1442 ! extracted from all the layers that has 'qv > 2*qvmin'. This fully
1443 ! preserves column moisture.
1444 if( dqv .gt. 1.e-20_r8 ) then
1445 sum = 0._r8
1446 do k = 1, mkx
1447 if( qv(i,k) .gt. 2._r8*qvmin ) sum = sum + qv(i,k)*dp(i,k)
1448 enddo
1449 aa = dqv*dp(i,1)/max(1.e-20_r8,sum)
1450 if( aa .lt. 0.5_r8 ) then
1451 do k = 1, mkx
1452 if( qv(i,k) .gt. 2._r8*qvmin ) then
1453 dum = aa*qv(i,k)
1454 qv(i,k) = qv(i,k) - dum
1455 qvten(i,k) = qvten(i,k) - dum/dt
1456 endif
1457 enddo
1458 else
1459 write(iulog,*) 'Full positive_moisture is impossible in vertical_diffusion'
1460 endif
1461 endif
1462 end do
