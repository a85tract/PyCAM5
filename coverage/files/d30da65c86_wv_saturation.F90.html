653! if q > qs(t) then tsp > t and qsp = qs(tsp) < q
654! if q < qs(t) then tsp < t and qsp = qs(tsp) > q
655!
656! Method:
657! a Newton method is used
658! first guess uses an algorithm provided by John Petch from the UKMO
659! we exclude points where the physical situation is unrealistic
660! e.g. where the temperature is outside the range of validity for the
661! saturation vapor pressure, or where the water vapor pressure
662! exceeds the ambient pressure, or the saturation specific humidity is
663! unrealistic
664!
665! Author: P. Rasch
666!
667!-----------------------------------------------------------------------
668!
669! input arguments
670!
671
672 real(r8), intent(in) :: q ! water vapor (kg/kg)
673 real(r8), intent(in) :: t ! temperature (K)
674 real(r8), intent(in) :: p ! pressure (Pa)
675 logical, intent(in) :: use_ice ! flag to include ice phase in calculations
676!
677! output arguments
678!
679 real(r8), intent(out) :: tsp ! saturation temp (K)
680 real(r8), intent(out) :: qsp ! saturation mixing ratio (kg/kg)
681 integer, intent(out) :: status ! flag representing state of output
682 ! 0 => Successful convergence
683 ! 1 => No calculation done: pressure or specific
684 ! humidity not within usable range
685 ! 2 => Run failed to converge
686 ! 4 => Temperature fell below minimum
687 ! 8 => Enthalpy not conserved
688!
689! local variables
690!
691 integer, parameter :: iter = 8 ! max number of times to iterate the calculation
692 integer :: l ! iterator
693
694 real(r8) es ! sat. vapor pressure
695 real(r8) gam ! change in sat spec. hum. wrt temperature (times hltalt/cpair)
696 real(r8) dgdt ! work variable
697 real(r8) g ! work variable
698 real(r8) hltalt ! lat. heat. of vap.
699 real(r8) qs ! spec. hum. of water vapor
700
701! work variables
702 real(r8) t1, q1, dt, dq
703 real(r8) qvd
704 real(r8) r1b, c1, c2
705 real(r8), parameter :: dttol = 1.e-4_r8 ! the relative temp error tolerance required to quit the iteration
706 real(r8), parameter :: dqtol = 1.e-4_r8 ! the relative moisture error tolerance required to quit the iteration
707 real(r8) enin, enout
708
709 ! Saturation specific humidity at this temperature
7104313712043137120 if (use_ice) then
7112156856021568560 call qsat(t, p, es, qs)
712 else
7132156856021568560 call qsat_water(t, p, es, qs)
714 end if
715
716 ! make sure a meaningful calculation is possible
71743137120171760602 if (p <= 5._r8*es .or. qs <= 0._r8 .or. qs >= 0.5_r8 &
7184287449442874494 .or. t < tmin .or. t > tmax) then
719262626262626 status = 1
720 ! Keep initial parameters when conditions aren't suitable
721 tsp = t
722 qsp = q
723 enin = 1._r8
724 enout = 1._r8
725
726 return
727 end if
728
729 ! Prepare to iterate
7304287449442874494 status = 2
731
732 ! Get initial enthalpy
733 if (use_ice) then
7342145668721456687 call calc_hltalt(t,hltalt)
735 else
7362141780721417807 call no_ip_hltalt(t,hltalt)
737 end if
7384287449442874494 enin = tq_enthalpy(t, q, hltalt)
739
740 ! make a guess at the wet bulb temp using a UKMO algorithm (from J. Petch)
741 c1 = hltalt*c3
742 c2 = (t + 36._r8)**2
743 r1b = c2/(c2 + c1*qs)
744 qvd = r1b * (q - qs)
745 tsp = t + ((hltalt/cpair)*qvd)
746
747 ! Generate qsp, gam, and enout from tsp.
748 if (use_ice) then
7492145668721456687 call qsat(tsp, p, es, qsp, gam=gam, enthalpy=enout)
750 else
7512141780721417807 call qsat_water(tsp, p, es, qsp, gam=gam, enthalpy=enout)
752 end if
753
754 ! iterate on first guess
7554691815789792651 do l = 1, iter
756
7578979265189792651 g = enin - enout
758 dgdt = -cpair * (1 + gam)
759
760 ! New tsp
761 t1 = tsp - g/dgdt
762 dt = abs(t1 - tsp)/t1
763 tsp = t1
764
765 ! bail out if past end of temperature range
766 if ( tsp < tmin ) then
76700 tsp = tmin
768 ! Get latent heat and set qsp to a value
769 ! that preserves enthalpy.
770 if (use_ice) then
77100 call calc_hltalt(tsp,hltalt)
772 else
77300 call no_ip_hltalt(tsp,hltalt)
774 end if
77500 qsp = (enin - cpair*tsp)/hltalt
77600 enout = tq_enthalpy(tsp, qsp, hltalt)
777 status = 4
778 exit
779 end if
780
781 ! Re-generate qsp, gam, and enout from new tsp.
7828979265189792651 if (use_ice) then
7834544084045440840 call qsat(tsp, p, es, q1, gam=gam, enthalpy=enout)
784 else
7854435181144351811 call qsat_water(tsp, p, es, q1, gam=gam, enthalpy=enout)
786 end if

788 qsp = q1
789
790 ! if converged at this point, exclude it from more iterations
7916452767364527673 if (dt < dttol .and. dq < dqtol) then
7924287449442874494 status = 0
793 exit
794 endif
795 end do
796
797 ! Test for enthalpy conservation
7984287449442874494 if (abs((enin-enout)/(enin+enout)) > 1.e-4_r8) status = 8
