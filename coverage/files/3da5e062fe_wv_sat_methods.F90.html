417! Old CAM implementation, also labelled Goff & Gratch (1946)
418
419! The water formula differs only due to compiler-dependent order of
420! operations, so differences are roundoff level, usually 0.
421
422! The ice formula gives fairly close answers to the current
423! implementation, but has been rearranged, and uses the
424! 1 atm melting point of water as the triple point.
425! Differences are thus small but above roundoff.
426
427! A curious fact: although using the melting point of water was
428! probably a mistake, it mildly improves accuracy for ice svp,
429! since it compensates for a systematic error in Goff & Gratch.
430
431elemental function OldGoffGratch_svp_water(t) result(es)
432 real(r8), intent(in) :: t
433 real(r8) :: es
434 real(r8) :: ps, e1, e2, f1, f2, f3, f4, f5, f
435
436 ps = 1013.246_r8
437 e1 = 11.344_r8*(1.0_r8 - t/tboil)
438 e2 = -3.49149_r8*(tboil/t - 1.0_r8)
439 f1 = -7.90298_r8*(tboil/t - 1.0_r8)
440 f2 = 5.02808_r8*log10(tboil/t)
441 f3 = -1.3816_r8*(10.0_r8**e1 - 1.0_r8)/10000000.0_r8
