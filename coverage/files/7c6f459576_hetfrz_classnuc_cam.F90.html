918 real(r8), parameter :: bc_num_to_mass = 4.669152e+17_r8 ! #/kg from emission
919 real(r8), parameter :: dst1_num_to_mass = 3.484e+15_r8 ! #/kg for dust in accumulation mode
920
921 real(r8) :: dmc, ssmc
922
923 real(r8) :: as_so4, as_du, as_soa
924 real(r8) :: dst1_num_imm, dst3_num_imm, bc_num_imm
925 real(r8) :: dmc_imm, ssmc_imm
926 real(r8) :: as_bc, as_pom, as_ss
927
928 real(r8) :: r_bc ! model radii of BC modes [m]
929 real(r8) :: r_dust_a1, r_dust_a3 ! model radii of dust modes [m]
930
931 integer :: i
932 real(r8) :: dst1_scale
933 !-------------------------------------------------------------------------------
934
935 ! init output vars
936 total_aer_num = 0._r8
937 total_interstial_aer_num = 0._r8
938 total_cloudborne_aer_num = 0._r8
939 coated_aer_num = 0._r8
940 uncoated_aer_num = 0._r8
941 hetraer = 0._r8
942 awcam = 0._r8
943 awfacm = 0._r8
944 dstcoat = 0._r8
945 na500 = 0._r8
946 tot_na500 = 0._r8
947
948 !*****************************************************************************
949 ! calculate intersitial aerosol
950 !*****************************************************************************
951
952 if (nmodes == MAM3_nmodes) then
953
954 if (.not. num_to_mass_in) then
955
956 as_so4 = aer(ii,kk,so4_accum)
957 as_bc = aer(ii,kk,bc_accum)
958 as_pom = aer(ii,kk,pom_accum)
959 as_soa = aer(ii,kk,soa_accum)
960 as_ss = aer(ii,kk,ncl_accum)
961 as_du = aer(ii,kk,dst_accum)
962
963 if (as_du > 0._r8) then
964 dst1_num = as_du/(as_so4+as_bc+as_pom+as_soa+as_ss+as_du) &
965 * aer(ii,kk,num_accum)*1.0e-6_r8 ! #/cm^3
966 else
967 dst1_num = 0.0_r8
968 end if
969
970 if (as_bc > 0._r8) then
971 bc_num = as_bc/(as_so4+as_bc+as_pom+as_soa+as_ss+as_du) &
972 * aer(ii,kk,num_accum)*1.0e-6_r8 ! #/cm^3
973 else
974 bc_num = 0.0_r8
975 end if
976
977 else
978
979 dst1_num = aer(ii,kk,dst_accum) * dst1_num_to_mass*1.0e-6_r8 ! #/cm^3, dust # in accumulation mode
980 bc_num = aer(ii,kk,bc_accum) * bc_num_to_mass*1.0e-6_r8 ! #/cm^3
981 end if
982 dmc = aer(ii,kk,dst_coarse)
983 ssmc = aer(ii,kk,ncl_coarse)
984
985 if (dmc > 0._r8 ) then
986 dst3_num = dmc/(ssmc+dmc) * aer(ii,kk,num_coarse)*1.0e-6_r8 ! #/cm^3
987 else
988 dst3_num = 0.0_r8
989 end if
990
991 else if (nmodes == MAM7_nmodes) then
992 bc_num = (aer(ii,kk,bc_accum)+aer(ii,kk,bc_pcarbon)) * bc_num_to_mass*1.0e-6_r8 ! #/cm^3
993 dst1_num = aer(ii,kk,num_finedust)*1.0e-6_r8 ! #/cm^3
994 dst3_num = aer(ii,kk,num_coardust)*1.0e-6_r8 ! #/cm^3
995 end if
996
997 !*****************************************************************************
998 ! calculate cloud borne aerosol
999 !*****************************************************************************
1000
1001 if (nmodes == MAM3_nmodes) then
1002
1003 as_so4 = aer_cb(ii,kk,so4_accum)
1004 as_bc = aer_cb(ii,kk,bc_accum)
1005 as_pom = aer_cb(ii,kk,pom_accum)
1006 as_soa = aer_cb(ii,kk,soa_accum)
1007 as_ss = aer_cb(ii,kk,ncl_accum)
1008 as_du = aer_cb(ii,kk,dst_accum)
1009
1010 if (as_du > 0._r8) then
1011 dst1_num_imm = as_du/(as_so4+as_bc+as_pom+as_soa+as_ss+as_du) &
1012 * aer_cb(ii,kk,num_accum)*1.0e-6_r8 ! #/cm^3
1013 else
1014 dst1_num_imm = 0.0_r8
1015 end if
1016
1017 if (as_bc > 0._r8) then
1018 bc_num_imm = as_bc/(as_so4+as_bc+as_pom+as_soa+as_ss+as_du) &
1019 * aer_cb(ii,kk,num_accum)*1.0e-6_r8 ! #/cm^3
1020 else
1021 bc_num_imm = 0.0_r8
1022 end if
1023
1024 dmc_imm = aer_cb(ii,kk,dst_coarse)
1025 ssmc_imm = aer_cb(ii,kk,ncl_coarse)
1026
1027 if (dmc_imm > 0._r8) then
1028 dst3_num_imm = dmc_imm/(ssmc_imm+dmc_imm) * aer_cb(ii,kk,num_coarse)*1.0e-6_r8 ! #/cm^3
1029 else
1030 dst3_num_imm = 0.0_r8
1031 end if
1032
1033 else if (nmodes == MAM7_nmodes) then
1034 ! primary carbon mode is insoluble and thus don't consider its cloud-borne state
1035 as_so4 = aer_cb(ii,kk,so4_accum)
1036 as_bc = aer_cb(ii,kk,bc_accum)
1037 as_pom = aer_cb(ii,kk,pom_accum)
1038 as_soa = aer_cb(ii,kk,soa_accum)
1039 as_ss = aer_cb(ii,kk,ncl_accum)
1040 if (as_bc > 0._r8) then
1041 bc_num_imm = as_bc/(as_so4+as_bc+as_pom+as_soa+as_ss) &
1042 * aer_cb(ii,kk,num_accum)*1.0e-6_r8 ! #/cm^3
1043 else
1044 bc_num_imm = 0.0_r8
1045 end if
1046 dst1_num_imm = aer_cb(ii,kk,num_finedust)*1.0e-6_r8 ! #/cm^3
1047 dst3_num_imm = aer_cb(ii,kk,num_coardust)*1.0e-6_r8 ! #/cm^3
1048 end if
1049
1050 total_interstial_aer_num(1) = bc_num
1051 total_interstial_aer_num(2) = dst1_num
1052 total_interstial_aer_num(3) = dst3_num
1053
1054 total_cloudborne_aer_num(1) = bc_num_imm
1055 total_cloudborne_aer_num(2) = dst1_num_imm
1056 total_cloudborne_aer_num(3) = dst3_num_imm
1057
1058 !*****************************************************************************
1059 ! calculate mass mean radius
1060 !*****************************************************************************
1061
1062 if (nmodes == MAM3_nmodes) then
1063
1064 if (aer(ii,kk,bc_accum)*1.0e-3_r8 > 1.0e-30_r8 .and. bc_num > 1.0e-3_r8) then
1065 r_bc = ( 3._r8/(4*pi*specdens_bc)*aer(ii,kk,bc_accum)/(bc_num*1.0e6_r8) )**(1._r8/3._r8)
1066 else
1067 r_bc = 0.04e-6_r8
1068 end if
1069
1070 if (aer(ii,kk,dst_accum)*1.0e-3_r8 > 1.0e-30_r8 .and. dst1_num > 1.0e-3_r8) then
1071 r_dust_a1 = ( 3._r8/(4*pi*specdens_dust)*aer(ii,kk,dst_accum)/(dst1_num*1.0e6_r8) )**(1._r8/3._r8)
1072 else
1073 r_dust_a1 = 0.258e-6_r8
1074 end if
1075
1076 if (aer(ii,kk,dst_coarse)*1.0e-3_r8 > 1.0e-30_r8 .and. dst3_num > 1.0e-3_r8) then
1077 r_dust_a3 = ( 3._r8/(4*pi*specdens_dust)*aer(ii,kk,dst_coarse)/(dst3_num*1.0e6_r8) )**(1._r8/3._r8)
1078 else
1079 r_dust_a3 = 1.576e-6_r8
1080 end if
1081
1082 else if (nmodes == MAM7_nmodes) then
1083
1084 if ((aer(ii,kk,bc_accum)+aer(ii,kk,bc_pcarbon))*1.0e-3_r8 > 1.0e-30_r8 &
1085 .and. bc_num > 1.0e-3_r8) then
1086 r_bc = ( 3._r8/(4*pi*specdens_bc)*(aer(ii,kk,bc_accum)+aer(ii,kk,bc_pcarbon))/ &
1087 (bc_num*1.0e6_r8) )**(1._r8/3._r8)
1088 else
1089 r_bc = 0.067e-6_r8 ! from emission size
1090 end if
1091
1092 if (aer(ii,kk,dst_finedust)*1.0e-3_r8 > 1.0e-30_r8 .and. dst1_num > 1.0e-3_r8) then
1093 r_dust_a1 = ( 3._r8/(4*pi*specdens_dust)*aer(ii,kk,dst_finedust)/(dst1_num*1.0e6_r8) )**(1._r8/3._r8)
1094 else
1095 r_dust_a1 = 0.258e-6_r8
1096 end if
1097
1098 if (aer(ii,kk,dst_coardust)*1.0e-3_r8 > 1.0e-30_r8 .and. dst3_num > 1.0e-3_r8) then
1099 r_dust_a3 = ( 3._r8/(4*pi*specdens_dust)*aer(ii,kk,dst_coardust)/(dst3_num*1.0e6_r8) )**(1._r8/3._r8)
1100 else
1101 r_dust_a3 = 1.576e-6_r8
1102 end if
1103 end if
1104
1105 hetraer(1) = r_bc
1106 hetraer(2) = r_dust_a1
1107 hetraer(3) = r_dust_a3
1108
1109 !*****************************************************************************
1110 ! calculate coated fraction
1111 !*****************************************************************************
1112
1113 if (nmodes == MAM3_nmodes) then
1114
1115 fac_volsfc_bc = exp(2.5_r8*alnsg_mode_accum**2)
1116 fac_volsfc_dust_a1 = exp(2.5_r8*alnsg_mode_accum**2)
1117 fac_volsfc_dust_a3 = exp(2.5_r8*alnsg_mode_coarse**2)
1118
1119 vol_shell(2) = ( aer(ii,kk,so4_accum)/specdens_so4 + &
1120 aer(ii,kk,pom_accum)*pom_equivso4_factor/specdens_pom + &
1121 aer(ii,kk,soa_accum)*soa_equivso4_factor/specdens_soa )/rhoair
1122
1123 vol_core(2) = aer(ii,kk,dst_accum)/(specdens_dust*rhoair)
1124
1125 ! ratio1 = vol_shell/vol_core =
1126 ! actual hygroscopic-shell-volume/dust-core-volume
1127 ! ratio2 = 6.0_r8*dr_so4_monolayers_pcage/(dgncur_a*fac_volsfc_dust)
1128 ! = (shell-volume corresponding to n_so4_monolayers_pcage)/core-volume
1129 ! The 6.0/(dgncur_a*fac_volsfc_dust) = (mode-surface-area/mode-volume)
1130 ! Note that vol_shell includes both so4, pom, AND soa as "equivalent so4",
1131 ! The soa_equivso4_factor accounts for the lower hygroscopicity of soa.
1132 !
1133 ! Define xferfrac_pcage = min( 1.0, ratio1/ratio2)
1134 ! But ratio1/ratio2 == tmp1/tmp2, and coding below avoids possible overflow
1135
1136 ! bc
1137 vol_shell(1) = vol_shell(2)
1138 vol_core(1) = aer(ii,kk,bc_accum)/(specdens_bc*rhoair)
1139 tmp1 = vol_shell(1)*(r_bc*2._r8)*fac_volsfc_bc
1140 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(1), 0.0_r8)
1141 dstcoat(1) = tmp1/tmp2
1142
1143 ! dust_a1
1144 tmp1 = vol_shell(2)*(r_dust_a1*2._r8)*fac_volsfc_dust_a1
1145 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(2), 0.0_r8)
1146 dstcoat(2) = tmp1/tmp2
1147
1148 ! dust_a3
1149 vol_shell(3) = aer(ii,kk,so4_coarse)/(specdens_so4*rhoair)
1150 vol_core(3) = aer(ii,kk,dst_coarse)/(specdens_dust*rhoair)
1151 tmp1 = vol_shell(3)*(r_dust_a3*2._r8)*fac_volsfc_dust_a3
1152 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(3), 0.0_r8)
1153 dstcoat(3) = tmp1/tmp2
1154
1155 else if (nmodes == MAM7_nmodes) then
1156
1157 ! for BC, only consider primary carbon mode,
1158 ! because most of particles in this mode are uncoated
1159 ! and nearly all particles in accumulation mode are coated
1160 fac_volsfc_bc = exp(2.5_r8*alnsg_mode_pcarbon**2)
1161
1162 vol_shell(1) = ( aer(ii,kk,pom_pcarbon)*pom_equivso4_factor/specdens_pom )/rhoair
1163 vol_core(1) = aer(ii,kk,bc_pcarbon)/(specdens_bc*rhoair)
1164 tmp1 = vol_shell(1)*(r_bc*2._r8)*fac_volsfc_bc
1165 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(1), 0.0_r8)
1166 dstcoat(1) = tmp1/tmp2
1167
1168 fac_volsfc_dust_a1 = exp(2.5_r8*alnsg_mode_finedust**2)
1169 fac_volsfc_dust_a3 = exp(2.5_r8*alnsg_mode_coardust**2)
1170
1171 vol_shell(2) = aer(ii,kk,so4_finedust)/(specdens_so4*rhoair)
1172 vol_core(2) = aer(ii,kk,dst_finedust)/(specdens_dust*rhoair)
1173
1174 tmp1 = vol_shell(2)*(r_dust_a1*2._r8)*fac_volsfc_dust_a1
1175 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(2), 0.0_r8)
1176 dstcoat(2) = tmp1/tmp2
1177
1178 vol_shell(3) = aer(ii,kk,so4_coardust)/(specdens_so4*rhoair)
1179 vol_core(3) = aer(ii,kk,dst_coardust)/(specdens_dust*rhoair)
1180 tmp1 = vol_shell(3)*(r_dust_a3*2._r8)*fac_volsfc_dust_a3
1181 tmp2 = max(6.0_r8*dr_so4_monolayers_dust*vol_core(3), 0.0_r8)
1182 dstcoat(3) = tmp1/tmp2
1183
1184 end if
1185
1186 if (dstcoat(1) > 1._r8) dstcoat(1) = 1._r8
1187 if (dstcoat(1) < 0.001_r8) dstcoat(1) = 0.001_r8
1188 if (dstcoat(2) > 1._r8) dstcoat(2) = 1._r8
1189 if (dstcoat(2) < 0.001_r8) dstcoat(2) = 0.001_r8
1190 if (dstcoat(3) > 1._r8) dstcoat(3) = 1._r8
1191 if (dstcoat(3) < 0.001_r8) dstcoat(3) = 0.001_r8
1192
1193 do i = 1, 3
1194 total_aer_num(i) = total_interstial_aer_num(i) + total_cloudborne_aer_num(i)
1195 coated_aer_num(i) = total_interstial_aer_num(i)*dstcoat(i)
1196 uncoated_aer_num(i) = total_interstial_aer_num(i)*(1._r8-dstcoat(i))
1197 end do
1198
1199 if (nmodes == MAM7_nmodes) then
1200 coated_aer_num(1) = (aer(ii,kk,bc_pcarbon)*bc_num_to_mass*1.0e-6_r8)*dstcoat(1)+ &
1201 (aer(ii,kk,bc_accum)*bc_num_to_mass*1.0e-6_r8)
1202 uncoated_aer_num(1) = (aer(ii,kk,bc_pcarbon)*bc_num_to_mass*1.0e-6_r8)*(1._r8-dstcoat(1))
1203 end if
1204
1205 if (nmodes == MAM3_nmodes) then
1206 dst1_scale = 0.488_r8 ! scaled for D>0.5-1 um from 0.1-1 um
1207 else if (nmodes == MAM7_nmodes) then
1208 dst1_scale = 0.566_r8 ! scaled for D>0.5-2 um from 0.1-2 um
1209 end if
1210
1211 tot_na500 = total_aer_num(1)*0.0256_r8 & ! scaled for D>0.5 um using Clarke et al., 1997; 2004; 2007: rg=0.1um, sig=1.6
1212 + total_aer_num(2)*dst1_scale + total_aer_num(3)
1213
1214 na500 = total_interstial_aer_num(1)*0.0256_r8 & ! scaled for D>0.5 um using Clarke et al., 1997; 2004; 2007: rg=0.1um, sig=1.6
1215 + total_interstial_aer_num(2)*dst1_scale + total_interstial_aer_num(3)
1216
1217 !*****************************************************************************
1218 ! prepare some variables for water activity
1219 !*****************************************************************************
1220
1221 if (nmodes == MAM3_nmodes) then
1222
1223 ! accumulation mode for dust_a1
1224 if (aer(ii,kk,num_accum) > 0._r8) then
1225 awcam(2) = (dst1_num*1.0e6_r8)/aer(ii,kk,num_accum)* &
1226 ( aer(ii,kk,so4_accum) + aer(ii,kk,soa_accum) + &
1227 aer(ii,kk,pom_accum) + aer(ii,kk,bc_accum) )*1.0e9_r8 ! [mug m-3]
1228 else
1229 awcam(2) = 0._r8
1230 end if
1231
1232 if (awcam(2) > 0._r8) then
1233 awfacm(2) = ( aer(ii,kk,bc_accum) + aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) )/ &
1234 ( aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) + aer(ii,kk,so4_accum) + aer(ii,kk,bc_accum) )
1235 else
1236 awfacm(2) = 0._r8
1237 end if
1238
1239 ! accumulation mode for bc
1240 if (aer(ii,kk,num_accum) > 0._r8) then
1241 awcam(1) = (bc_num*1.0e6_r8)/aer(ii,kk,num_accum)* &
1242 ( aer(ii,kk,so4_accum) + aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) + aer(ii,kk,bc_accum) )*1.0e9_r8 ! [mug m-3]
1243 else
1244 awcam(1) = 0._r8
1245 end if
1246 awfacm(1) = awfacm(2)
1247
1248 ! coarse mode for dust_a3
1249 if (aer(ii,kk,num_coarse) > 0._r8) then
1250 awcam(3) = (dst3_num*1.0e6_r8)/aer(ii,kk,num_coarse)* aer(ii,kk,so4_coarse)*1.0e9_r8
1251 else
1252 awcam(3) = 0._r8
1253 end if
1254 awfacm(3) = 0._r8
1255
1256 else if (nmodes == MAM7_nmodes) then
1257
1258 ! accumulation mode for bc (primary carbon mode is insoluble)
1259 if (aer(ii,kk,num_accum) > 0._r8) then
1260 awcam(1) = (bc_num*1.0e6_r8)/aer(ii,kk,num_accum)* &
1261 ( aer(ii,kk,so4_accum) + aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) + aer(ii,kk,bc_accum) )*1.0e9_r8 ! [mug m-3]
1262 else
1263 awcam(1) = 0._r8
1264 end if
1265
1266 if (awcam(1) > 0._r8) then
1267 awfacm(1) = ( aer(ii,kk,bc_accum) + aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) )/ &
1268 ( aer(ii,kk,soa_accum) + aer(ii,kk,pom_accum) + aer(ii,kk,so4_accum) + aer(ii,kk,bc_accum) )
1269 else
1270 awfacm(1) = 0._r8
1271 end if
1272
1273 if (aer(ii,kk,num_finedust) > 0._r8) then
1274 awcam(2) = (dst1_num*1.0e6_r8)/aer(ii,kk,num_finedust)* aer(ii,kk,so4_finedust)*1.0e9_r8
1275 else
1276 awcam(2) = 0._r8
1277 end if
1278 awfacm(2) = 0._r8
1279
1280 if (aer(ii,kk,num_coardust) > 0._r8) then
1281 awcam(3) = (dst3_num*1.0e6_r8)/aer(ii,kk,num_coardust)* aer(ii,kk,so4_coardust)*1.0e9_r8
1282 else
1283 awcam(3) = 0._r8
1284 end if
1285 awfacm(3) = 0._r8
1286
1287 end if
1288
1289end subroutine get_aer_num
