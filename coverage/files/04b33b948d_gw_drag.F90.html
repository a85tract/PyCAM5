1566 ! Output temperature tendencies due to diffusion and from kinetic energy.
1567 call outfld(trim(prefix)//'TTGWSDF', dttdf / cpair, ncol, lchnk)
1568 call outfld(trim(prefix)//'TTGWSKE', dttke / cpair, ncol, lchnk)
1569
1570
1571 ! Output tau broken down into zonal and meridional components.
1572
1573 taux = 0._r8
1574 tauy = 0._r8
1575
1576 ! Project c, and convert each component to a wavenumber index.
1577 ! These are mappings from the wavenumber index of tau to those of taux
1578 ! and tauy, respectively.
1579 do l=-band%ngwv,band%ngwv
1580 ix(:,l) = c_to_l(c(:,l)*xv)
1581 iy(:,l) = c_to_l(c(:,l)*yv)
1582 end do
1583
1584 ! Find projection of tau.
1585 do k = 1, pver
1586 do l = -band%ngwv,band%ngwv
1587 do i = 1, ncol
1588 taux(i,ix(i,l),k) = taux(i,ix(i,l),k) &
1589 + abs(tau(i,l,k)*xv(i))
1590 tauy(i,iy(i,l),k) = tauy(i,iy(i,l),k) &
1591 + abs(tau(i,l,k)*yv(i))
1592 end do
1593 end do
1594 end do
1595
1596 do l=-band%ngwv,band%ngwv
1597
1598 dummyx = taux(:,l,:)
1599 dummyy = tauy(:,l,:)
1600
1601 dumc1x = tau_fld_name(l, prefix, x_not_y=.true.)
1602 dumc1y = tau_fld_name(l, prefix, x_not_y=.false.)
1603
1604 call outfld(dumc1x,dummyx,ncol,lchnk)
1605 call outfld(dumc1y,dummyy,ncol,lchnk)
1606
1607 enddo
1608
1609
1610 ! Output momentum flux in each cardinal direction.
1611 mf = 0._r8
1612
1613 do k = 1, pver
1614
1615 ! Convert wind speed components to wavenumber indices.
1616 iu = c_to_l(u(:,k))
1617 iv = c_to_l(v(:,k))
1618
1619 ! Sum tau components in each cardinal direction.
1620 ! Split west/east and north/south based on whether wave speed exceeds
1621 ! wind speed.
1622 do l = -band%ngwv, band%ngwv
1623
1624 where (iu > l)
1625 mf(:,k,west) = mf(:,k,west) + taux(:,l,k)
1626 elsewhere
1627 mf(:,k,east) = mf(:,k,east) + taux(:,l,k)
1628 end where
1629
1630 where (iv > l)
1631 mf(:,k,south) = mf(:,k,south) + tauy(:,l,k)
1632 elsewhere
1633 mf(:,k,north) = mf(:,k,north) + tauy(:,l,k)
1634 end where
1635
1636 end do
1637
1638 end do
1639
1640 call outfld(trim(prefix)//'WMF',mf(:,:,west),ncol,lchnk)
1641 call outfld(trim(prefix)//'EMF',mf(:,:,east),ncol,lchnk)
1642 call outfld(trim(prefix)//'SMF',mf(:,:,south),ncol,lchnk)
1643 call outfld(trim(prefix)//'NMF',mf(:,:,north),ncol,lchnk)
1644
1645 ! Simple output fields written to history file.
1646 ! Total wind tendencies.
1647 call outfld (trim(prefix)//'UTGWSPEC', utgw , ncol, lchnk)
1648 call outfld (trim(prefix)//'VTGWSPEC', vtgw , ncol, lchnk)
1649 call outfld (trim(prefix)//'TTGWSPEC', ttgw , ncol, lchnk)
1650
1651 ! Tau in each direction.
1652 call outfld (trim(prefix)//'TAUE', taucd(:,:,east), ncol, lchnk)
1653 call outfld (trim(prefix)//'TAUW', taucd(:,:,west), ncol, lchnk)
1654 call outfld (trim(prefix)//'TAUN', taucd(:,:,north), ncol, lchnk)
1655 call outfld (trim(prefix)//'TAUS', taucd(:,:,south), ncol, lchnk)
1656
1657 call outfld (trim(prefix)//'TAUNET', taucd(:,:,east)+taucd(:,:,west), &
1658 ncol, lchnk)
1659
1660contains
