791 function parametric_coordinates(sphere, corners3D,ref_map_in, corners,u2qmap,facenum) result (ref)
792 implicit none
793 type (spherical_polar_t), intent(in) :: sphere
794 type (cartesian2D_t) :: ref
795
796 type (cartesian3D_t) :: corners3D(4) !x,y,z coords of element corners
797 integer,optional :: ref_map_in ! default is global variable 'cubed_sphere_map'
798 ! optional arguments, only needed for ref_map=1 (equi-angle gnomonic projection):
799 type (cartesian2D_t),optional :: corners(4) ! gnomonic coords of element corners
800 real (kind=real_kind),optional :: u2qmap(4,2)
801 integer,optional :: facenum
802
803
804 ! local
805 integer :: i, MAX_NR_ITER=10
806 real(kind=real_kind) :: D(2,2),Dinv(2,2),detD,a,b,resa,resb,dela,delb,costh
807 real(kind=real_kind) :: tol_sq=1e-26
808 type (spherical_polar_t) :: sphere1, sphere_tmp
809 integer :: ref_map
810
811 !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
812 ! newton iteration on: ref=ref - df^-1 (ref2sphere(ref) - sphere)
813 !
814 ! Generic version written in terms of HOMME's 'ref2sphere' and 'Dmap' operaters,
815 ! with no assumption as to the type of map (gnomonic, equi-angular, parametric)
816 !
817 ! f = ref2sphere(xvec) - sphere
818 ! df = d(ref2sphere)
819 !
820 ! D = diag(cos(theta),1) * d(ref2sphere) d(ref2sphere) = diag(1/cos(theta),1)*D
821 !
822 ! df = diag(1/cos(theta),1)*D
823 ! df^-1 = D^-1 * diag(cos(theta),1)
824 !
825 !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
826 if (present(ref_map_in)) then
827 ref_map=ref_map_in
828 else
829 ref_map=cubed_sphere_map
830 endif
831 costh=cos(sphere%lat)
832 a=0
833 b=0
834 i=0
835 do
836 sphere1 = ref2sphere(a,b,corners3D,ref_map,corners,facenum)
837 resa = sphere1%lon - sphere%lon
838 if (resa>dd_pi) resa=resa-2*dd_pi
839 if (resa<-dd_pi) resa=resa+2*dd_pi
840
841 resb = sphere1%lat - sphere%lat
842
843 call Dmap(D,a,b,corners3D,ref_map,corners,u2qmap,facenum)
844 detD = D(1,1)*D(2,2) - D(1,2)*D(2,1)
845 Dinv(1,1) = D(2,2)/detD
846 Dinv(1,2) = -D(1,2)/detD
847 Dinv(2,1) = -D(2,1)/detD
848 Dinv(2,2) = D(1,1)/detD
849
850 dela = Dinv(1,1)*costh*resa + Dinv(1,2)*resb
851 delb = Dinv(2,1)*costh*resa + Dinv(2,2)*resb
852 a = a - dela
853 b = b - delb
854 i=i+1
855 if ( (costh*resa)**2 + resb**2 < tol_sq .or. MAX_NR_ITER < i) exit
856 end do
857 ref%x=a
858 ref%y=b
859
860 end function parametric_coordinates

1231 !
1232 !

1234 implicit none

1236 type (parallel_t) , intent(in) :: par
1237 type (interpdata_t) , intent(out) :: interpdata(:)

1242 integer countx, missing_pts,ierr
1243 integer :: npts_mult_claims,max_claims
1244
1245 real (kind=real_kind) :: dp,latdeg(nlat+1),clat(nlat+1),w(nlat+1),w_staggered(nlat)
1246 real (kind=real_kind) :: clat_staggered(nlat),latdeg_st(nlat),err,err2

1251
1252 type (quadrature_t) :: gp
1253
1254
1255 logical,save :: first_time=.TRUE.

1257 ! Array to make sure each interp point is on exactly one process
1258 type (cartesian2D_t),allocatable :: cart_vec(:,:)
1259 integer :: k

1271 endif

1275 enddo

1286 endif

1288 if(size(gweight)>0)deallocate(gweight)
1289 endif

1301 lon(i)=2*dd_pi*(i-1)/nlon
1302 enddo

1306 end do

1308 endif

1314 end do
1315 endif

1317 do j=1,nlat
1318 lat(j) = -dd_pi/2 + dd_pi*(j-.5d0)/nlat
1319 end do

1321 endif
1322
1323 if (gridtype==1 .or. gridtype==3) then
1324 ! gridtype=1 plat=nlat gweight(1:nlat)=w(1:plat)
1325 ! gridtype=3 plat=nlat+1 gweight(1:nlat)=w_staggered(1:plat-1)
1326
1327 ! L-R dynamics uses a regular latitude distribution (not gausian).
1328 ! The algorithm below is a bastardized version of LSM: map.F.
1329 dp = 180d0/(plat-1)
1330 do j = 1, plat
1331 latdeg(j) = -90d0 + (j-1)*dp
1332 clat(j) = latdeg(j)*dd_pi/180d0
1333 end do
1334
1335 ! Calculate latitudes for the staggered grid
1336
1337 do j = 1, plat-1
1338 clat_staggered(j) = (clat(j) + clat(j+1)) / 2

1340 end do

1343 ! For a sanity check, the sum of w across all lats should be 2, or 1 across
1344 ! half of the latitudes.
1345
1346 do j = 2, plat-1
1347 w(j) = sin(clat_staggered(j)) - sin(clat_staggered(j-1))
1348 end do
1349 w(1) = sin(clat_staggered(1)) + 1
1350 w(plat) = w(1)
1351

1353 if (abs(sum(w(1:plat)) - 2) > 1e-8) then
1354 write(iulog,*) 'interpolate_mod: w weights do not sum to 2. sum=',sum(w(1:plat))
1355 call abortmp('interpolate_mod: weights do not sum to 2.')
1356 end if
1357
1358 dp = dd_pi / (plat-1)
1359 do j = 1, plat-1
1360 w_staggered(j) = sin(clat(j+1)) - sin(clat(j))
1361 end do
1362
1363
1364 if (abs(sum(w_staggered(1:plat-1)) - 2) > 1e-8) then
1365 write(iulog,*) 'interpolate_mod: staggered weights do not sum to 2. sum=',sum(w_staggered(1:plat-1))
1366 call abortmp('interpolate_mod: weights do not sum to 2.')
1367 end if

1370 gweight(1:nlat)=w(1:plat)
1371 endif
1372 if (gridtype==3) then
1373 gweight(1:nlat)=w_staggered(1:plat-1)
1374 endif
1375 endif
1376
1377
1378 ! go through once, counting the number of points on each element
1379 sphere%r=1
1380 local_elem_num = -1
1381 local_elem_gid = -1
1382 global_elem_gid = -1
1383 err=0
1384 do j=1,nlat
1385 do i=1,nlon
1386 sphere%lat=lat(j)
1387 sphere%lon=lon(i)
1388
1389 !debug=.false.
1390 !if (j==18 .and. i==95) debug=.true.
1391 number = -1
1392 if ( (cubed_sphere_map /= 0) .or. MeshUseMeshFile) then
1393 call cube_facepoint_unstructured(sphere, cart, number, elem)

1396 if (cart%x + 1.0d0.le.0.0d0) cart%x = -1.0d0
1397 if (cart%x - 1.0d0.ge.0.0d0) cart%x = 1.0d0
1398 if (cart%y + 1.0d0.le.0.0d0) cart%y = -1.0d0
1399 if (cart%y - 1.0d0.ge.0.0d0) cart%y = 1.0d0
1400
1401 local_elem_num(j,i) = number
1402 local_elem_gid(j,i) = elem(number)%vertex%number
1403 cart_vec(j,i) = cart ! local element coordiante of interpolation point
1404 endif
1405 else
1406 call cube_facepoint_ne(sphere, ne, cart, number)
1407 ! the sphere point belongs to the element number on face = face_no.
1408 ! do I own this element?
1409 if (number /= -1) then
1410 do ii=1,nelemd
1411 if (number == elem(ii)%vertex%number) then
1412 local_elem_gid(j,i) = number
1413 local_elem_num(j,i) = ii
1414 cart_vec(j,i) = cart ! local element coordinate found above
1415 exit
1416 endif
1417 enddo
1418 endif
1419 endif
1420 ii=local_elem_num(j,i)
1421 if (ii /= -1) then
1422 ! compute error: map 'cart' back to sphere and compare with original
1423 ! interpolation point:
1424 sphere2_xyz = spherical_to_cart( ref2sphere(cart%x,cart%y, &
1425 elem(ii)%corners3D,cubed_sphere_map,elem(ii)%corners,elem(ii)%facenum ))
1426 sphere_xyz = spherical_to_cart(sphere)
1427 err=max(err,distance(sphere2_xyz,sphere_xyz))
1428 endif
1429 enddo
1430 if (par%masterproc) then
1431 if ((MOD(j,64).eq.1).or.(j.eq.nlat)) then
1432 print *,'finished latitude ',j,' of ',nlat
1433 endif
1434 endif
1435 enddo
1436 err2=err
1437#ifdef _MPI
1438 call MPI_Allreduce(err,err2,1,MPIreal_t,MPI_MAX,par%comm,ierr)
1439#endif
1440 if (par%masterproc) then
1441 write(iulog,'(a,e12.4)') 'Max interpolation point search error: ',err2
1442 endif

1445 global_elem_gid = local_elem_gid
1446#ifdef _MPI
1447 call MPI_Allreduce(local_elem_gid, global_elem_gid, nlat*nlon, MPIinteger_t, MPI_MAX, par%comm,ierr)
1448#endif

1452 do i=1,nlon

1460 endif
1461 end do
1462 end do
1463
1464 countx=maxval(interpdata(1:nelemd)%n_interp)

1494 allocate(interpdata(ii)%interp_xy( ngrid ) )

1496 allocate(interpdata(ii)%ilon( ngrid ) )
1497 interpdata(ii)%n_interp=0 ! reset counter
1498 enddo
1499 do j=1,nlat
1500 do i=1,nlon
1501 if (local_elem_gid(j,i) == global_elem_gid(j,i) .and. &
1502 local_elem_gid(j,i) /= -1 ) then
1503 ii = local_elem_num(j,i)
1504 ngrid = interpdata(ii)%n_interp + 1

1507 interpdata(ii)%ilon( ngrid ) = i
1508 interpdata(ii)%ilat( ngrid ) = j
1509 endif
1510 enddo
1511 enddo
1512
1513 ! now lets compute the number of points that were claimed by
1514 ! more than one element:
1515 do j=1,nlat
1516 do i=1,nlon
1517 if (local_elem_gid(j,i) == -1) then
1518 local_elem_gid(j,i)=0
1519 else
1520 local_elem_gid(j,i)=1
1521 endif
1522 enddo
1523 enddo
1524 global_elem_gid = local_elem_gid
1525#ifdef _MPI
1526 call MPI_Allreduce(local_elem_gid, global_elem_gid, nlat*nlon, MPIinteger_t, MPI_SUM, par%comm,ierr)
1527#endif
1528 if (par%masterproc) then
1529 countx=0
1530 do j=1,nlat
1531 do i=1,nlon
1532 if (global_elem_gid(j,i)>1) countx=countx+1
1533 enddo
1534 enddo
1535 npts_mult_claims=countx
1536 max_claims=maxval(global_elem_gid)
1537 endif
1538
1539 if (par%masterproc) then
1540 print *,'Number of interpolation points claimed by more than one element: ',npts_mult_claims
1541 print *,'max number of elements which claimed the same interpolation point:',max_claims
1542 endif
1543
1544 deallocate(global_elem_gid)
1545 deallocate(local_elem_num)
1546 deallocate(local_elem_gid)
1547 deallocate(cart_vec)
1548
1549 ! check if every point in interpolation grid was claimed by an element:

1551 count_total = nlat*nlon
1552 if(par%masterproc) then
1553 write(iulog,"(3A,I4,A,I7,a,i5)")"Error:",__FILE__," ",__LINE__," count_total:",count_total," missing:",missing_pts

1556 call abortmp('Error: interpolation points not claimed by any element')
