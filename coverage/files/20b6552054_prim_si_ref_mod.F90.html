117 subroutine prim_si_refmat_init(lprint,masterproc,refstate)
118 use physical_constants , only : rgas
119
120 logical, intent(in) :: lprint ! print/no print of ref matrices
121 logical, intent(in) :: masterproc ! is the master process?
122 type(ref_state_t) :: refstate ! reference state structure
123
124 integer k,l,kk,kkk ! level indices
125 integer ik1,ik2,nkk ! misc. integers
126
127 ! =================================================================
128 ! Integration matrices of hydrostatic equation(href) and conversion
129 ! term(a). href computed as in ccm0 but isothermal bottom ecref
130 ! calculated to conserve energy
131 ! =================================================================
132
133 do k=1,plev
134 do l=1,plev
135 refstate%Href(l,k) = 0.0_r8
136 refstate%Cref(l,k) = 0.0_r8
137 end do
138 end do
139
140 ! ======================================================================
141 ! Mean atmosphere energy conversion term is consistent with continiuty
142 ! equation. In Cref, 1st index = column; 2nd index = row of matrix.
143 ! Mean atmosphere energy conversion term is energy conserving
144 ! ======================================================================
145
146 do k=1,plev
147 refstate%Cref(k,k) = 0.50_r8/refstate%hypm(k)
148 do l=1,k-1
149 refstate%Cref(l,k) = 1.0_r8/refstate%hypm(k)
150 end do
151 end do
152
153 ! ====================================================================
154 ! Reference hydrostatic integration matrix consistent with conversion
155 ! term for energy conservation. In href, 1st index = column;
156 ! 2nd index = row of matrix.
157 ! ====================================================================
158
159 do k = 1,plev
160 do l = k,plev
161 refstate%Href(l,k) = refstate%Cref(k,l)*refstate%hypd(l)
162 end do
163 end do
164
165 ! ==================================
166 ! Print statements
167 ! ==================================
168
169 if (lprint.and.masterproc) then
170 nkk = plev/13
171 if (mod(plev,13).ne.0) nkk = nkk + 1
172 write(iulog,*)' '
173 write(iulog,*)'INITCOM: Hydrostatic matrix href'
174 do kk=1,nkk
175 ik1 = 1 + (kk-1)*13
176 ik2 = min0( ik1+12, plev )
177 write(iulog,9920) (k,k=ik1,ik2)
178 do kkk=1,plev
179 write(iulog,9910) kkk,(refstate%Href(kkk,k),k=ik1,ik2)
180 end do
181 end do
182 write(iulog,*)' '
183 write(iulog,*)'INITCOM: Thermodynamic matrix ecref'
184 do kk=1,nkk
185 ik1 = 1 + (kk-1)*13
186 ik2 = MIN( ik1+12, plev )
187 write(iulog,9920) (k,k=ik1,ik2)
188 do kkk=1,plev
189 write(iulog,9910) kkk,(refstate%Cref(kkk,k),k=ik1,ik2)
190 end do
191 end do
192 end if
193
194 ! =======================
195 ! Multiply href by r
196 ! =======================
197
198 do k=1,plev
199 do kk=1,plev
200 refstate%Href(kk,k) = refstate%Href(kk,k)*Rgas
201 end do
202 end do
203

2059910 format( 1x,i3,13f9.5)
