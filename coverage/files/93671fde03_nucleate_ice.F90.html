4! Purpose:
5! A parameterization of ice nucleation.
6!
7! *** This module is intended to be a "portable" code layer. Ideally it should
8! *** not contain any use association of modules that belong to the model framework.
9!
10!
11! Method:
12! The current method is based on Liu & Penner (2005) & Liu et al. (2007)
13! It related the ice nucleation with the aerosol number, temperature and the
14! updraft velocity. It includes homogeneous freezing of sulfate & immersion
15! freezing on mineral dust (soot disabled) in cirrus clouds, and
16! Meyers et al. (1992) deposition nucleation in mixed-phase clouds
17!
18! The effect of preexisting ice crystals on ice nucleation in cirrus clouds is included,
19! and also consider the sub-grid variability of temperature in cirrus clouds,
20! following X. Shi et al. ACP (2014).
21!
22! Ice nucleation in mixed-phase clouds now uses classical nucleation theory (CNT),
23! follows Y. Wang et al. ACP (2014), Hoose et al. (2010).
24!
25! Authors:
26! Xiaohong Liu, 01/2005, modifications by A. Gettelman 2009-2010
27! Xiangjun Shi & Xiaohong Liu, 01/2014.
28!
29! With help from C. C. Chen and B. Eaton (2014)
30!-------------------------------------------------------------------------------

453SUBROUTINE Vpreice(P_in, T_in, R_in, C_in, S_in, V_out)
454
455 ! based on Karcher et al. (2006)
456 ! VERTICAL VELOCITY CALCULATED FROM DEPOSITIONAL LOSS TERM
457
458 ! SUBROUTINE arguments
459 REAL(r8), INTENT(in) :: P_in ! [Pa],INITIAL AIR pressure
460 REAL(r8), INTENT(in) :: T_in ! [K] ,INITIAL AIR temperature
461 REAL(r8), INTENT(in) :: R_in ! [m],INITIAL MEAN ICE CRYSTAL NUMBER RADIUS
462 REAL(r8), INTENT(in) :: C_in ! [m-3],INITIAL TOTAL ICE CRYSTAL NUMBER DENSITY, [1/cm3]
463 REAL(r8), INTENT(in) :: S_in ! [-],INITIAL ICE SATURATION RATIO;; if <1, use hom threshold Si
464 REAL(r8), INTENT(out) :: V_out ! [m/s], VERTICAL VELOCITY REDUCTION (caused by preexisting ice)
465
466 ! SUBROUTINE parameters
467 REAL(r8), PARAMETER :: ALPHAc = 0.5_r8 ! density of ice (g/cm3), !!!V is not related to ALPHAc
468 REAL(r8), PARAMETER :: FA1c = 0.601272523_r8
469 REAL(r8), PARAMETER :: FA2c = 0.000342181855_r8
470 REAL(r8), PARAMETER :: FA3c = 1.49236645E-12_r8
471 REAL(r8), PARAMETER :: WVP1c = 3.6E+10_r8
472 REAL(r8), PARAMETER :: WVP2c = 6145.0_r8
473 REAL(r8), PARAMETER :: FVTHc = 11713803.0_r8
474 REAL(r8), PARAMETER :: THOUBKc = 7.24637701E+18_r8
475 REAL(r8), PARAMETER :: SVOLc = 3.23E-23_r8 ! SVOL=XMW/RHOICE
476 REAL(r8), PARAMETER :: FDc = 249.239822_r8
477 REAL(r8), PARAMETER :: FPIVOLc = 3.89051704E+23_r8
478 REAL(r8) :: T,P,S,R,C
479 REAL(r8) :: A1,A2,A3,B1,B2
480 REAL(r8) :: T_1,PICE,FLUX,ALP4,CISAT,DLOSS,VICE
481
482 T = T_in ! K , K
483 P = P_in*1e-2_r8 ! Pa , hpa
484
485 IF (S_in.LT.1.0_r8) THEN
486 S = 2.349_r8 - (T/259.0_r8) ! homogeneous freezing threshold, according to Ren & McKenzie, 2005
487 ELSE
488 S = S_in ! INPUT ICE SATURATION RATIO, -, >1
489 ENDIF
490
491 R = R_in*1e2_r8 ! m => cm
492 C = C_in*1e-6_r8 ! m-3 => cm-3
493 T_1 = 1.0_r8/ T
494 PICE = WVP1c * EXP(-(WVP2c*T_1))
495 ALP4 = 0.25_r8 * ALPHAc
496 FLUX = ALP4 * SQRT(FVTHc*T)
497 CISAT = THOUBKc * PICE * T_1
498 A1 = ( FA1c * T_1 - FA2c ) * T_1
499 A2 = 1.0_r8/ CISAT
500 A3 = FA3c * T_1 / P
501 B1 = FLUX * SVOLc * CISAT * ( S-1.0_r8 )
502 B2 = FLUX * FDc * P * T_1**1.94_r8
503 DLOSS = FPIVOLc * C * B1 * R**2 / ( 1.0_r8+ B2 * R )
504 VICE = ( A2 + A3 * S ) * DLOSS / ( A1 * S ) ! 2006,(19)
505 V_out = VICE*1e-2_r8 ! cm/s => m/s
506
507END SUBROUTINE Vpreice

522 real(r8) :: Sihom, deta
523 integer :: i
524
525 Sihom = 2.349_r8-Tmean/259.0_r8 ! homogeneous freezing threshold, according to Ren & McKenzie, 2005
526 fhom = 0.0_r8
527
528 do i = Nbin, 1, -1
529
530 deta = (i - 0.5_r8 - Nbin/2)*6.0_r8/Nbin ! PDF_T=0 outside (Tmean-3*detaT, Tmean+3*detaT)
531 Sbin(i) = RHimean*exp(deta*detaT*seta/Tmean**2.0_r8)
532 PDF_T(i) = exp(-deta**2.0_r8/2.0_r8)*6.0_r8/(sqrt(2.0_r8*Pi)*Nbin)
533
534
535 if (Sbin(i).ge.Sihom) then
536 fhom = fhom + PDF_T(i)
537 else
538 exit
539 end if
540 end do
541
542 fhom=fhom/0.997_r8 ! accounting for the finite limits (-3 , 3)
543
544end subroutine frachom
