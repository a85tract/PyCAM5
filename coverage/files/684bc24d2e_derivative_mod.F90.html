2450
2451 ! The GLL (Gauss-Lobatto-Legendre) points are from [-1,1],
2452 ! we have a bunch of sub-intervals defined by intervals that
2453 ! go from [a,b] so we need to linearly map [-1,1] -> [a,b]
2454 ! all the GLL points by y = (a/2)(1-x) + (b/2)(1+x)
2455 do i=1,intervals
2456 a = -one + (i-one)*two/intervals
2457 b = -one + i *two/intervals
2458 sub_gll(i,:) = (a+b)/two + gll%points(:)/intervals
2459 end do
2460
2461 ! Now to interpolate from the values at the input GLL
2462 ! points to the sub-GLL points. Do this by Lagrange
2463 ! interpolation. The jth Lagrange interpolating polynomial
2464 ! for points x_i is
2465 ! \prod_{i\ne j} (x-x_i)/(x_j-x_i)
2466 ! These are then multiplied by the sampled values y_i
2467 ! and summed.
2468
2469 ! Save some time by pre-computing the denominitor. I think
2470 ! this is OK since all the points are of order 1 so should
2471 ! be well behaved.
2472 do n = 1,np
2473 x_j = gll%points(n)
2474 x = one
2475 do m = 1,np
2476 if (m.ne.n) then
2477 x_i = gll%points(m)
2478 x = x * (x_j-x_i)
2479 endif
2480 end do
2481 legrange_div(n)= x
2482 end do
2483 do i=1,intervals
2484 do n=1,np
2485 x = sub_gll(i,n)
2486 do j = 1,np
2487 y = one
2488 do m = 1,np
2489 if (m.ne.j) then
2490 x_i = gll%points(m)
2491 y = y * (x-x_i)
2492 end if
2493 end do
2494 Lagrange_interp(i,n,j) = y/legrange_div(j)
2495 end do
2496 end do
2497 end do
2498
2499 ! Integration is the GLL weights times Jacobians times
2500 ! the interpolated values:
2501 ! w^t I Y I^t w
2502 ! where
2503 ! w is GLL weights and Jacobians,
2504 ! I is the Lagrange_interp matrix, and
2505 ! Y is the coefficient matrix, sampled_val.
2506 ! This can be written J Y J^t where
2507 ! J = w^t I
2508 ! J is integration_matrix
2509 do i=1,intervals
2510 integration_matrix(i,:) = MATMUL(gll%weights(:),Lagrange_interp(i,:,:))
2511 end do
2512
2513 ! There is still the Jacobian to consider. We are
2514 ! integrating over [a,b] x [c,d] where
2515 ! |b-a| = |d-c| = 2/Intervals
2516 ! Multiply the weights appropriately given that
2517 ! they are defined for a 2x2 square
2518 integration_matrix = integration_matrix/intervals
2519
2520 end subroutine allocate_subcell_integration_matrix
