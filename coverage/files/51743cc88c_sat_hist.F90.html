591 subroutine read_next_position( ncols )
592 use time_manager, only: get_curr_date, get_prev_date
593 use time_manager, only: set_time_float_from_date
594
595 implicit none
596
597 integer, intent(out) :: ncols
598
599 integer :: ierr
600 integer :: yr, mon, day, tod
601 real(r8) :: begdatetime, enddatetime
602 integer :: beg_ndx, end_ndx, i
603
604 real(r8) :: datetime
605
606 call get_curr_date(yr, mon, day, tod)
607 call set_time_float_from_date(begdatetime, yr, mon, day, tod-half_step*sathist_ntimestep)
608 call set_time_float_from_date(enddatetime, yr, mon, day, tod+half_step*sathist_ntimestep)
609
610 ncols = 0
611
612 if ( first_datetime > enddatetime ) then
613 if (masterproc) write(iulog,'(a,2f16.6)') &
614 'sat_hist->read_next_position: all of the satellite date times are after the time window', first_datetime, enddatetime
615 return
616 endif
617 if ( last_datetime < begdatetime ) then
618 if (masterproc) write(iulog,'(a,2f16.6)') &
619 'sat_hist->read_next_position: all of the satellite date times are before the time window', begdatetime, last_datetime
620 return
621 endif
622
623 call t_startf ('sat_hist::read_next_position')
624
625 beg_ndx = -99
626 end_ndx = -99
627
628 bnds_loop: do i = time_ndx,n_profiles
629
630 call read_buffered_datetime( datetime, i )
631
632 if ( datetime>begdatetime .and. beg_ndx<0 ) beg_ndx = i
633 if ( datetime>enddatetime ) exit bnds_loop
634 end_ndx = i
635 enddo bnds_loop
636
637 if (beg_ndx == -99 .and. end_ndx== -99) then
638 if (masterproc) write(iulog,'(a)') 'sat_hist->read_next_position: must be beyond last position -- returning.'
639 return
640 endif
641
642 ! Advance the search forward, but because of ntimesteps, it is possible
643 ! for observations used here to be used again. However, we should not go
644 ! back before the previous beginning time.
645 if (beg_ndx>0) time_ndx = beg_ndx
646
647 ncols = end_ndx-beg_ndx+1
648 if (ncols > 0) then
649 allocate( obs_lats(ncols), obs_lons(ncols) )
650 in_start_col = beg_ndx
651
652 ierr = pio_get_var( infile, lat_vid, (/beg_ndx/), (/ncols/), obs_lats )
653 ierr = pio_get_var( infile, lon_vid, (/beg_ndx/), (/ncols/), obs_lons )
654
655 endif
656
657 call t_stopf ('sat_hist::read_next_position')
658 end subroutine read_next_position

671 real(r8), intent(in) :: mod_lons(ncols * sathist_nclosest)
672 real(r8), intent(in) :: mod_dists(ncols * sathist_nclosest)
673 integer, intent(in) :: nfils

676 integer :: yr, mon, day ! year, month, and day components of a date
677 integer :: nstep ! current timestep number
678 integer :: ncdate ! current date in integer format [yyyymmdd]

738 endif

740 ierr = copy_data( infile, orbit_vid, tape%File, out_orbid, in_start_col, nfils, ncols )
741 endif
742 if (prof_vid>0) then
743 ierr = copy_data( infile, prof_vid, tape%File, out_profid, in_start_col, nfils, ncols )
744 endif
745 if (zenith_vid>0) then
746 ierr = copy_data( infile, zenith_vid, tape%File, out_zenithid, in_start_col, nfils, ncols )
747 endif
748 if (in_julian_vid>0) then
749 ierr = copy_data( infile, in_julian_vid, tape%File, out_julian_vid, in_start_col, nfils, ncols )
750 endif
751 if (in_occ_type_vid>0) then
752 ierr = copy_data( infile, in_occ_type_vid, tape%File, out_occ_type_vid, in_start_col, nfils, ncols )

755 ierr = copy_data( infile, in_localtime_vid, tape%File, out_localtime_vid, in_start_col, nfils, ncols )

761 call t_stopf ('sat_hist::write_record_coord')

767 subroutine get_indices( lats, lons, ncols, nocols, has_dyn_flds, col_ndxs, chk_ndxs, &
768 fdyn_ndxs, ldyn_ndxs, phs_owners, dyn_owners, mlats, mlons, phs_dists )
769
770 use dyn_grid, only : dyn_grid_get_colndx
771 use phys_grid, only: get_rlat_p, get_rlon_p
772
773 integer, intent(in) :: ncols
774 real(r8), intent(in) :: lats(ncols)
775 real(r8), intent(in) :: lons(ncols)
776 integer, intent(in) :: nocols
777 logical, intent(in) :: has_dyn_flds
778 integer, intent(out) :: col_ndxs(nocols)
779 integer, intent(out) :: chk_ndxs(nocols)
780 integer, intent(out) :: fdyn_ndxs(nocols)
781 integer, intent(out) :: ldyn_ndxs(nocols)
782 integer, intent(out) :: phs_owners(nocols)
783 integer, intent(out) :: dyn_owners(nocols)
784 real(r8), intent(out) :: mlats(nocols)
785 real(r8), intent(out) :: mlons(nocols)
786 real(r8), intent(out) :: phs_dists(nocols)
787
788 integer :: i, j, ndx
789 real(r8) :: lat, lon
790
791 integer, allocatable :: ichks(:),icols(:),idyn1s(:),idyn2s(:), iphs_owners(:), idyn_owners(:)
792 real(r8), allocatable :: rlats(:), rlons(:), plats(:), plons(:), iphs_dists(:)
793
794 integer :: gcols(sathist_nclosest)
795
796 call t_startf ('sat_hist::get_indices')
797
798 allocate(ichks(sathist_nclosest),icols(sathist_nclosest),idyn1s(sathist_nclosest), &
799 idyn2s(sathist_nclosest),iphs_owners(sathist_nclosest),idyn_owners(sathist_nclosest))
800 allocate(rlats(sathist_nclosest), rlons(sathist_nclosest), plats(sathist_nclosest), &
801 plons(sathist_nclosest), iphs_dists(sathist_nclosest) )
802
803 col_ndxs = -1
804 chk_ndxs = -1
805 fdyn_ndxs = -1
806 ldyn_ndxs = -1
807 phs_owners = -1
808 dyn_owners = -1
809 phs_dists = -1
810
811 ndx = 0
812 do i = 1,ncols
813
814 lat = lats(i)
815 lon = lons(i)
816
817 if ( lon >= 360._r8) then
818 lon = lon-360._r8
819 endif
820 if ( lon < 0._r8) then
821 lon = lon+360._r8
822 endif
823 if (lat<-90._r8 .or. lat>90._r8) then
824 write(iulog,*) 'sat_hist::get_indices lat = ',lat
825 call endrun('sat_hist::get_indices : lat must be between -90 and 90 degrees (-90<=lat<=90)')
826 endif
827 if (lon<0._r8 .or. lon>=360._r8) then
828 write(iulog,*) 'sat_hist::get_indices lon = ',lon
829 call endrun('sat_hist::get_indices : lon must be between 0 and 360 degrees (0<=lon<360)')
830 endif
831
832 call find_cols( lat, lon, sathist_nclosest, iphs_owners, ichks, icols, &
833 gcols, iphs_dists, plats, plons )
834
835 if (has_dyn_flds) then
836 call dyn_grid_get_colndx( gcols, sathist_nclosest, idyn_owners, idyn1s, idyn2s )
837 endif
838
839 do j = 1, sathist_nclosest
840
841 if (debug .and. iam==iphs_owners(j) ) then
842 if ( abs(plats(j)-rlats(j))>1.e-3_r8 ) then
843 write(*,'(a,3f20.12)') ' lat, plat, rlat = ', lat, plats(j), rlats(j)
844 write(*,'(a,3f20.12)') ' lon, plon, rlon = ', lon, plons(j), rlons(j)
845 call endrun('sat_hist::get_indices: dyn lat is different than phys lat ')
846 endif
847 if ( abs(plons(j)-rlons(j))>1.e-3_r8 ) then
848 write(*,'(a,3f20.12)') ' lat, plat, rlat = ', lat, plats(j), rlats(j)
849 write(*,'(a,3f20.12)') ' lon, plon, rlon = ', lon, plons(j), rlons(j)
850 call endrun('sat_hist::get_indices: dyn lon is different than phys lon ')
851 endif
852 endif
853
854 ndx = ndx+1
855
856 chk_ndxs(ndx) = ichks(j)
857 col_ndxs(ndx) = icols(j)
858 fdyn_ndxs(ndx) = idyn1s(j)
859 ldyn_ndxs(ndx) = idyn2s(j)
860 mlats(ndx) = plats(j)
861 mlons(ndx) = plons(j)
862 phs_owners(ndx) = iphs_owners(j)
863 dyn_owners(ndx) = idyn_owners(j)
864 phs_dists(ndx) = iphs_dists(j)
865 enddo
866 enddo
867
868 deallocate(ichks, icols, idyn1s, idyn2s, iphs_owners, idyn_owners)
869 deallocate(rlats, rlons, plats, plons, iphs_dists )
870
871 call t_stopf ('sat_hist::get_indices')
872 end subroutine get_indices

961 subroutine find_cols(lat, lon, nclosest, owner, lcid, icol, gcol, distmin, mlats, mlons)
962 use physconst, only: rearth
963 use phys_grid, only: get_rlon_all_p, get_rlat_all_p, get_gcol_p, get_ncols_p
964 use spmd_utils, only: iam, npes, mpi_integer, mpi_real8, mpicom
965
966 real(r8),intent(in) :: lat, lon ! requested location in degrees
967 integer, intent(in) :: nclosest ! number of closest points to find
968 integer, intent(out) :: owner(nclosest) ! rank of chunk owner
969 integer, intent(out) :: lcid(nclosest) ! local chunk index
970 integer, intent(out) :: icol(nclosest) ! column index within the chunk
971 integer, intent(out) :: gcol(nclosest) ! global column index
972 real(r8),intent(out) :: distmin(nclosest) ! the distance (m) of the closest column(s)
973 real(r8),intent(out) :: mlats(nclosest) ! the latitude of the closest column(s)
974 real(r8),intent(out) :: mlons(nclosest) ! the longitude of the closest column(s)
975
976 real(r8) :: dist
977 real(r8) :: rlats(pcols), rlons(pcols)
978 real(r8) :: latr, lonr
979
980 integer :: my_owner(nclosest)
981 integer :: my_lcid(nclosest)
982 integer :: my_icol(nclosest)
983 integer :: my_gcol(nclosest)
984 real(r8) :: my_distmin(nclosest)
985 real(r8) :: my_mlats(nclosest)
986 real(r8) :: my_mlons(nclosest)
987
988 integer :: c, i, j, k, ierr, ncols, mindx(1)
989 real(r8) :: sendbufr(3)
990 real(r8) :: recvbufr(3,npes)
991 integer :: sendbufi(4)
992 integer :: recvbufi(4,npes)
993
994 call t_startf ('sat_hist::find_cols')
995
996 latr = lat/rad2deg ! to radians
997 lonr = lon/rad2deg ! to radians
998
999 my_owner(:) = -999
1000 my_lcid(:) = -999
1001 my_icol(:) = -999
1002 my_gcol(:) = -999
1003 my_mlats(:) = -999
1004 my_mlons(:) = -999
1005 my_distmin(:) = 1.e10_r8
1006
1007 chk_loop: do c=begchunk,endchunk
1008 ncols = get_ncols_p(c)
1009 call get_rlat_all_p(c, pcols, rlats)
1010 call get_rlon_all_p(c, pcols, rlons)
1011
1012 col_loop: do i = 1,ncols
1013 ! Use the Spherical Law of Cosines to find the great-circle distance.
1014 dist = acos(sin(latr) * sin(rlats(i)) + cos(latr) * cos(rlats(i)) * cos(rlons(i) - lonr)) * rearth
1015
1016 closest_loop: do j = nclosest, 1, -1
1017 if (dist < my_distmin(j)) then
1018
1019 if (j < nclosest) then
1020 my_distmin(j+1) = my_distmin(j)
1021 my_owner(j+1) = my_owner(j)
1022 my_lcid(j+1) = my_lcid(j)
1023 my_icol(j+1) = my_icol(j)
1024 my_gcol(j+1) = my_gcol(j)
1025 my_mlats(j+1) = my_mlats(j)
1026 my_mlons(j+1) = my_mlons(j)
1027 end if
1028
1029 my_distmin(j) = dist
1030 my_owner(j) = iam
1031 my_lcid(j) = c
1032 my_icol(j) = i
1033 my_gcol(j) = get_gcol_p(c,i)
1034 my_mlats(j) = rlats(i) * rad2deg
1035 my_mlons(j) = rlons(i) * rad2deg
1036 else
1037 exit
1038 end if
1039 enddo closest_loop
1040
1041 enddo col_loop
1042 enddo chk_loop
1043
1044 k = 1
1045
1046 do j = 1, nclosest
1047
1048 sendbufr(1) = my_distmin(k)
1049 sendbufr(2) = my_mlats(k)
1050 sendbufr(3) = my_mlons(k)
1051
1052 call mpi_allgather( sendbufr, 3, mpi_real8, recvbufr, 3, mpi_real8, mpicom, ierr )
1053
1054 mindx = minloc(recvbufr(1,:))
1055 distmin(j) = recvbufr(1,mindx(1))
1056 mlats(j) = recvbufr(2,mindx(1))
1057 mlons(j) = recvbufr(3,mindx(1))
1058
1059 sendbufi(1) = my_owner(k)
1060 sendbufi(2) = my_lcid(k)
1061 sendbufi(3) = my_icol(k)
1062 sendbufi(4) = my_gcol(k)
1063
1064 call mpi_allgather( sendbufi, 4, mpi_integer, recvbufi, 4, mpi_integer, mpicom, ierr )
1065
1066 owner(j) = recvbufi(1,mindx(1))
1067 lcid(j) = recvbufi(2,mindx(1))
1068 icol(j) = recvbufi(3,mindx(1))
1069 gcol(j) = recvbufi(4,mindx(1))
1070
1071 if ( iam == owner(j) ) then
1072 k = k+1
1073 endif
1074
1075 enddo
1076
1077 call t_stopf ('sat_hist::find_cols')
1078
1079 end subroutine find_cols
